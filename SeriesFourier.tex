\subsubsection{Contexto Histórico}
\begin{itemize}
    \item $1750$: D. Bernoulli propone $f(x) = \sum_{n=1}^{\infty} a_n \sin(nx)$ para la ecuación de onda. Intervienen Euler, D'Alembert, Lagrange.
    \item $\simeq 1800 - 1807$: Aparece la representación $f(x) = \frac{a_0}{2} + \sum_{n=1}^{\infty} (a_n \cos(nx) + b_n \sin(nx))$.
    \item $1822$: ¿$a_n, b_n$?
\end{itemize}
Integrando en un periodo para hallar $a_0$:
\begin{align*}
    \int_{-\pi}^{\pi} f(x) dx &= \int_{-\pi}^{\pi} \left[ \frac{a_0}{2} + \sum_{n=1}^{\infty} (a_n \cos(nx) + b_n \sin(nx)) \right] dx \\
    &= \int_{-\pi}^{\pi} \frac{a_0}{2} dx + 0 = \pi a_0
\end{align*}
De manera análoga, multiplicando por $\cos(mx)$ e integrando:
\begin{align*}
    \int_{-\pi}^{\pi} f(x) \cos(mx) dx &= \int_{-\pi}^{\pi} \left[ \frac{a_0}{2} + \sum_{n=1}^{\infty} (a_n \cos(nx) + b_n \sin(nx)) \right] \cos(mx) dx \\
    &= \int_{-\pi}^{\pi} \frac{a_0}{2} \cos(mx) dx \\
    &\quad + \sum_{n=1}^{\infty} \left( a_n \int_{-\pi}^{\pi} \cos(nx) \cos(mx) dx + b_n \int_{-\pi}^{\pi} \sin(nx) \cos(mx) dx \right) \\
    &= 0 + a_m \int_{-\pi}^{\pi} \cos^2(mx) dx + 0 \\
    &= a_m \pi \implies a_m = \frac{1}{\pi} \int_{-\pi}^{\pi} f(x) \cos(mx) dx
\end{align*}

\subsection{Funciones Periódicas}

\begin{definicion}{Función Periódica}{func_per}
$f: \mathbb{R} \to \mathbb{C}$ es periódica si $\exists T > 0$ tal que $f(x) = f(x+T)$ para todo $x \in \mathbb{R}$. En este caso, $T$ es el Periodo de $f$ ($f$ es $T$-periódica).
\end{definicion}

\begin{nota}{Construcción de funciones periódicas}{constr_per}
Para construir una función $T$-periódica basta con definir $\tilde{f}: [0, T[ \to \mathbb{C}$ y extenderla:
$$ f(x) = \tilde{f}(x - kT) \quad \text{si } kT \le x < (k+1)T, \; k \in \mathbb{Z} $$
También podemos considerar $\tilde{f}: [-T/2, T/2[ \to \mathbb{C}$ y \"copiar\".

\textbf{Ejemplos:}
\begin{enumerate}
    \item $f(x) = 1$ para $0 \le x < 1$, y $f(x) = -1$ para $-1 \le x < 0$. ($2$-periódica).
    \item $f(x) = x$ para $0 \le x < 1$. ($1$-periódica).
    \item $f(x) = x$ para $-1/2 \le x < 1/2$. ($1$-periódica).
\end{enumerate}
\end{nota}

\begin{observacion}{Cambio de Periodo}{cambio_per}
Pregunta: ¿Puede descomponerse cualquier función periódica en términos de funciones periódicas elementales?

Nota: Supongamos que $f: \mathbb{R} \to \mathbb{C}$ es $T$-periódica (la pienso como $f: [-T/2, T/2[ \to \mathbb{C}$ y extendida periódica). Tomo $L > 0$ y defino $g: [-L/2, L/2[ \to \mathbb{C}$ como $g(x) = f\left(\frac{T}{L}x\right)$.
$$ -\frac{L}{2} \le x < \frac{L}{2} \implies -\frac{T}{2} \le \frac{T}{L}x < \frac{T}{2} $$
¡El periodo no es relevante! A partir de ahora todas las funciones serán $2\pi$-periódicas:
$f: [-\pi, \pi[ \to \mathbb{C}$ o $f: [0, 2\pi[ \to \mathbb{C}$.
\end{observacion}

\subsection{Coeficientes y Series de Fourier}

\begin{definicion}{Coeficientes de Fourier}{coef_fourier}
Dada $f: \mathbb{R} \to \mathbb{C}$ $2\pi$-periódica, se definen sus coeficientes de Fourier (si las integrales existen):
\begin{align*}
    a_n &= \frac{1}{\pi} \int_{-\pi}^{\pi} f(x) \cos(nx) dx \quad n = 0, 1, 2, \dots \\
    b_n &= \frac{1}{\pi} \int_{-\pi}^{\pi} f(x) \sin(nx) dx \quad n = 1, 2, 3, \dots
\end{align*}
\end{definicion}

\begin{nota}{Integración en un periodo}{int_per}
Si $g: \mathbb{R} \to \mathbb{C}$ es $T$-periódica, entonces $\int_{-T/2}^{T/2} g(x) dx = \int_0^T g(x) dx$ (Para la demostración basta con hacer cambio de variable). En la definición de los coeficientes de Fourier se puede tomar $\int_0^{2\pi}$ o $\int_I$, con $I$ de longitud $2\pi$.
\end{nota}

\begin{definicion}{Serie de Fourier}{serie_fourier}
Si $f: \mathbb{R} \to \mathbb{C}$ es $2\pi$-periódica, se define su serie de Fourier (serie formal):
$$ f \sim \frac{a_0}{2} + \sum_{n=1}^{\infty} \left( a_n \cos(nx) + b_n \sin(nx) \right) $$
\end{definicion}

\subsubsection{Convergencia}

\begin{nota}{Tipos de Convergencia}{conv}
En un espacio normado $(V, ||\cdot||) \implies (v_n)_n$ converge a $v$ si $||v_n - v|| \xrightarrow{n \to \infty} 0$.

Para funciones $g_n: \Omega \to \mathbb{C}$:
\begin{itemize}
    \item \textbf{Conv puntual:} $|g_n(x) - g(x)| \to 0 \quad \forall x \in \Omega$
    \item \textbf{Conv unif:} $\sup_{x \in \Omega} |g_n(x) - g(x)| \to 0 \implies$ Intercambiar (límites, integrales, derivadas).
\end{itemize}
\end{nota}
\subsection{Forma compleja de la serie de Fourier}

A partir de las identidades de Euler:
\begin{equation*}
    \cos(nt) = \frac{e^{int} + e^{-int}}{2}, \quad \sin(nt) = \frac{e^{int} - e^{-int}}{2i} = -i \frac{e^{int} - e^{-int}}{2}
\end{equation*}

Entonces la serie se desarrolla como:
\begin{align*}
    \frac{a_0}{2} + \sum_{n=1}^{\infty} (a_n \cos(nt) + b_n \sin(nt)) &= \frac{a_0}{2} + \sum_{n=1}^{\infty} \left( \frac{a_n}{2} e^{int} + \frac{a_n}{2} e^{-int} - \frac{b_n i}{2} e^{int} + \frac{b_n i}{2} e^{-int} \right) \\
    &= \frac{a_0}{2} + \sum_{n=1}^{\infty} \frac{a_n - b_n i}{2} e^{int} + \sum_{n=1}^{\infty} \frac{a_n + b_n i}{2} e^{-int}
\end{align*}

Para el último sumatorio, aplicando el cambio de variable $-n=k$, se obtiene $\sum_{-\infty}^{-1} \frac{a_{-k} + b_{-k}i}{2} e^{ikt}$.

\begin{nota}{Notación}{notacion_coeficientes}
Se definen los coeficientes complejos de la siguiente manera:
\begin{itemize}
    \item Cuando $n = 0$: $\hat{f}(0) = \frac{a_0}{2}$
    \item Cuando $n \ge 1$: $\hat{f}(n) = \frac{a_n - i b_n}{2}$
    \item Cuando $n \le -1$: $\hat{f}(n) = \frac{a_{-n} + i b_{-n}}{2}$
\end{itemize}
\end{nota}

Entonces, la fórmula compleja de la serie de Fourier queda expresada como:
\begin{equation*}
    f(t) \sim \sum_{n=-\infty}^{\infty} \hat{f}(n) e^{int}
\end{equation*}

\subsubsection{Coeficientes de la forma compleja}

Desarrollando la expresión para cada rango de $n$:
\begin{itemize}
    \item Para $n = 0$:
    \begin{equation*}
        \hat{f}(0) = \frac{a_0}{2} = \frac{1}{2\pi} \int_{-\pi}^{\pi} f(t) dt
    \end{equation*}
    \item Para $n \ge 1$:
    \begin{align*}
        \hat{f}(n) &= \frac{1}{2}(a_n - ib_n) = \frac{1}{2\pi} \int_{-\pi}^{\pi} f(t) (\cos(nt) - i \sin(nt)) dt \\
        &= \frac{1}{2\pi} \int_{-\pi}^{\pi} f(t) (\overline{\cos(nt) + i \sin(nt)}) dt \\
        &= \frac{1}{2\pi} \int_{-\pi}^{\pi} f(t) \overline{e^{int}} dt = \frac{1}{2\pi} \int_{-\pi}^{\pi} f(t) e^{-int} dt
    \end{align*}
    \item Para $n \le -1$:
    \begin{equation*}
        \hat{f}(n) = \frac{1}{2}(a_{-n} + i b_{-n}) = \frac{1}{2\pi} \int_{-\pi}^{\pi} f(t) (\cos(nt) + i \sin(nt)) dt = \frac{1}{2\pi} \int_{-\pi}^{\pi} f(t) e^{-int} dt
    \end{equation*}
\end{itemize}

Los coeficientes complejos quedan de la forma general unificada:
\begin{equation*}
    \hat{f}(n) = \frac{1}{2\pi} \int_{-\pi}^{\pi} f(t) e^{-int} dt, \quad \forall n \in \mathbb{Z}
\end{equation*}

\begin{observacion}{Recuperación de coeficientes reales}{recuperacion_coeficientes}
Si conocemos la sucesión $(\hat{f}(n))_{n \in \mathbb{Z}}$, ¿cómo encontramos $a_n$ y $b_n$?  

Sabemos que $a_0 = 2\hat{f}(0)$. Despejando en las ecuaciones para $n \ge 1$, obtenemos:
\begin{align*}
    a_n &= \hat{f}(n) + \hat{f}(-n) \\
    b_n &= i(\hat{f}(n) - \hat{f}(-n))
\end{align*}
\end{observacion}

\subsection{El grupo topológico $\mathbb{T}$ y el espacio $C(\mathbb{T})$}

\begin{definicion}{El toro $\mathbb{T}$}{toro}
Se define el toro como $\mathbb{T} = \{z \in \mathbb{C} : |z|=1\} = \partial \mathbb{D}$.
\end{definicion}

\begin{observacion}{Propiedades de $\mathbb{T}$}{prop_toro}
El toro unidimensional $\mathbb{T} = \{z \in \mathbb{C} : |z|=1\} = \partial\mathbb{D}$ cumple las siguientes propiedades fundamentales:
\begin{enumerate}
    \item Como espacio topológico, es un subconjunto cerrado y acotado dentro de $\mathbb{C}$. Por tanto, es un espacio compacto.
    \item Tiene estructura de \textbf{grupo abeliano} bajo la multiplicación de números complejos. Es decir, para cualesquiera $z_1, z_2 \in \mathbb{T}$, se cumple que $|z_1 \cdot z_2| = |z_1| \cdot |z_2| = 1 \cdot 1 = 1$, por lo que $z_1 \cdot z_2 \in \mathbb{T}$, además $z_1^{-1} \in \mathbb{T}$ y el elemento neutro es $1 \in \mathbb{T}$.
\end{enumerate}
\end{observacion}

\begin{nota}{El homomorfismo $\Phi$ e isomorfismo con $\mathbb{R} / 2\pi\mathbb{Z}$}{homomorfismo_toro}
Considérese la aplicación $\Phi: \mathbb{R} \to \mathbb{T}$ definida como $\Phi(t) = e^{it}$. Esta función cumple lo siguiente:
\begin{enumerate}
    \item \textbf{Es un homomorfismo de grupos:} Para cualesquiera $t_1, t_2 \in \mathbb{R}$, la exponencial compleja verifica que:
    \begin{equation*}
        \Phi(t_1 + t_2) = e^{i(t_1+t_2)} = e^{it_1}e^{it_2} = \Phi(t_1)\Phi(t_2)
    \end{equation*}
    \item \textbf{Es sobreyectivo:} Para todo $z \in \mathbb{T}$, existe al menos un $t \in \mathbb{R}$ tal que $z = e^{it}$ (basta con tomar $t = \arg(z)$).
    \item \textbf{Núcleo de la aplicación:} El núcleo (o \textit{ker}) está formado por todos los valores de $t$ que se aplican en el elemento neutro de $\mathbb{T}$ (que es el $1$):
    \begin{equation*}
        \ker \Phi = \{t \in \mathbb{R} : e^{it} = 1\} = \{2\pi k : k \in \mathbb{Z}\} = 2\pi\mathbb{Z}
    \end{equation*}
\end{enumerate}
Por el Primer Teorema de Isomorfía de grupos, al ser $\Phi$ un homomorfismo exhaustivo, el dominio cocientado por el núcleo es isomorfo a la imagen. Entonces:
\begin{equation*}
    \mathbb{T} \cong \mathbb{R} / 2\pi\mathbb{Z}
\end{equation*}
Como consecuencia directa de este isomorfismo, todo elemento del toro puede representarse de forma única en un periodo; es decir, para todo $z \in \mathbb{T}$, existe un único $t \in [-\pi, \pi[$ tal que $z = e^{it}$.
\end{nota}

\begin{nota}{Funciones en el Toro}{func_toro}
Supongamos que tenemos $f:\mathbb{T} \to \mathbb{C}$ y definimos $g:\mathbb{R} \to \mathbb{C}$ como $g(t) = f(e^{it})$. La función $g$ es $2\pi$-periódica.

Inversamente, si $g:\mathbb{R} \to \mathbb{C}$ es $2\pi$-periódica, se puede definir $f(e^{it}) = g(t)$ con $f:\mathbb{T} \to \mathbb{C}$.

Además, $f$ es continua si y solo si $g$ es continua. De igual forma, $f$ es medible si y solo si $g$ es medible.

A partir de aquí, utilizaremos la notación $f(z) = f(e^{it}) = f(t)$ indistintamente para $z \in \mathbb{T}$, $t \in [-\pi,\pi[$.
\end{nota}

\begin{definicion}{El espacio $C(\mathbb{T})$}{espacio_ct}
Se define el espacio de funciones continuas sobre el toro como $C(\mathbb{T}) = \{f:\mathbb{T} \to \mathbb{C} \mid f \text{ continua}\}$.

Esto es equivalente a $\{f:[-\pi,\pi] \to \mathbb{C} \mid f(\pi)=f(-\pi) \text{ y continua}\}$ y a $\{f:\mathbb{R} \to \mathbb{C} \mid 2\pi\text{-periódica y continua}\}$.

Además, dotado de la norma del supremo $\|f\|_\infty = \sup_{z \in \mathbb{T}} |f(z)| = \sup_{t \in [-\pi,\pi[} |f(t)|$, el espacio $(C(\mathbb{T}), \|\cdot\|_\infty)$ es un espacio vectorial normado.
\end{definicion}

\subsection{Polinomios Trigonométricos y Densidad}

\begin{definicion}{Polinomio trigonométrico}{pol_trig}
Un polinomio trigonométrico es una función de la forma:
\begin{equation*}
    f(t) = \sum_{n=M}^N c_n e^{int} = \frac{a_0}{2} + \sum_{n=1}^{\max(N,M)} (a_n \cos(nt) + b_n \sin(nt))
\end{equation*}
Denotamos al conjunto de polinomios trigonométricos como $\mathcal{P}(\mathbb{T})$.
\end{definicion}

\begin{observacion}{Recordatorio de Topología y Densidad}{recordatorio_densidad}
Supongamos un espacio topológico $X$, con $A, B \subseteq X$.
\begin{itemize}
    \item $A$ es denso $\iff \bar{A} = X$.
    \item Si $X$ es normado, $A$ es denso $\iff$ Fijado $x \in X, \forall \epsilon > 0 \ \exists y \in A \text{ tq } \|x - y\| < \epsilon$.
    \item Si $A$ es denso en $B$ y $B$ es denso en $C \implies A$ es denso en $C$.
    \item Si $A$ es denso en $C$ y $A \subseteq B \subseteq C \implies B$ es denso en $C$.
\end{itemize}

Si $C(K) = \{f: K \to \mathbb{C} \mid \text{continua}\}$, un subconjunto $A \subseteq C(K)$ cumple:
\begin{enumerate}
    \item $f_1, f_2 \in A \implies f_1 + f_2 \in A$
    \item $f_1, f_2 \in A \implies f_1 \cdot f_2 \in A$
\end{enumerate}
Entonces $A$ forma una \textbf{subálgebra}. Si además se cumple que:
\begin{enumerate}
    \setcounter{enumi}{2}
    \item $f \in A \implies \bar{f} \in A$
\end{enumerate}
Entonces $A$ es denso en $C(K)$.
\end{observacion}

\begin{proposicion}{Densidad en $C(\mathbb{T})$}{densidad_ct}
En nuestro caso $C(K) = C(\mathbb{T})$.

Si $f, g$ son polinomios trigonométricos:
\begin{align*}
    f(t) &= \sum_{n=-N_1}^{N_1} c_n e^{int} \\
    g(t) &= \sum_{n=-N_2}^{N_2} d_n e^{int}
\end{align*}
Se cumplen de manera directa las siguientes propiedades:
\begin{enumerate}
    \item $f + g \in \mathcal{P}(\mathbb{T})$
    \item $f \cdot g \in \mathcal{P}(\mathbb{T})$
    \item $\bar{f} = \sum_{n=-N_1}^{N_1} \overline{c_n e^{int}} = \sum_{n=-N_1}^{N_1} \overline{c_n} \cdot e^{-int} \in \mathcal{P}(\mathbb{T})$
\end{enumerate}
Por tanto, $\mathcal{P}(\mathbb{T})$ es denso en $C(\mathbb{T})$.
\end{proposicion}

\begin{nota}{Interpretación del Teorema de Densidad}{interpretacion_prop}
El resultado de la proposición anterior quiere decir:

Para toda $f \in C(\mathbb{T})$ y todo $\epsilon > 0$, existe un $P \in \mathcal{P}(\mathbb{T})$ (polinomio trigonométrico) tal que:
\begin{equation*}
    \sup_{t \in [-\pi, \pi]} |P(t) - f(t)| \le \epsilon
\end{equation*}


Es decir, $\|f - P\|_\infty \le \epsilon$.
\end{nota}

\begin{definicion}{El espacio $L_p(\mathbb{T})$}{espacio_lp}
Para $1 \le p < \infty$, decimos que una función $2\pi$-periódica $f:\mathbb{R} \to \mathbb{C}$ pertenece al espacio $L_p(\mathbb{T})$ si es medible en el sentido de Lebesgue y la integral de su valor absoluto a la potencia $p$ es finita. Formalmente:
\begin{equation*}
    f \in L_p(\mathbb{T}) \iff \text{medible Lebesgue y } \underbrace{\left( \frac{1}{2\pi} \int_{-\pi}^{\pi} |f(t)|^p dt \right)^{1/p}}_{\|f\|_p} < \infty
\end{equation*}



\end{definicion}

\begin{nota}{Propiedad en casi todas partes (c.p.p.)}{cpp}
Decimos que una propiedad matemática se cumple \textbf{casi por todas partes} (c.p.p. o \textit{a.e.} en inglés) si el conjunto de puntos donde no se verifica es tan insignificante que tiene medida de Lebesgue igual a cero.
\end{nota}

\begin{observacion}{Clases de equivalencia}{clases_eq}
En la práctica funcional, no distinguimos entre funciones que difieren en un conjunto de medida cero. Trabajamos sistemáticamente con clases de equivalencia, asumiendo que $f = g$ si coinciden casi por todas partes ($f \sim g$).
\end{observacion}

\begin{teorema}{Desigualdad de Hölder}{holder}
Imaginemos dos funciones $f \in L_r(\mathbb{T})$ y $g \in L_{r'}(\mathbb{T})$, donde los exponentes son conjugados (cumplen la relación $\frac{1}{r} + \frac{1}{r'} = 1$). Entonces, su producto $f \cdot g$ siempre será integrable ($L_1$), y su norma quedará acotada por el producto de sus normas respectivas:
\begin{equation*}
    \frac{1}{2\pi} \int_{-\pi}^{\pi} |f(t)g(t)| dt \le \left( \frac{1}{2\pi} \int_{-\pi}^{\pi} |f(t)|^r dt \right)^{1/r} \left( \frac{1}{2\pi} \int_{-\pi}^{\pi} |g(t)|^{r'} dt \right)^{1/r'}
\end{equation*}
De forma más compacta, esto se escribe como $\|fg\|_1 \le \|f\|_r \cdot \|g\|_{r'}$.
\end{teorema}

\begin{proposicion}{Inclusión de espacios $L_p$ y densidad}{incl_dens_lp}
Se cumplen las siguientes propiedades fundamentales para los espacios $L_p(\mathbb{T})$:
\begin{enumerate}
    \item \textbf{Inclusión de espacios:} Si un exponente es estrictamente menor que otro ($p < q$), el espacio $L_q$ está totalmente contenido en $L_p$. Es decir, si $\|f\|_q < \infty$, entonces $\|f\|_p < \infty$.
    
    \item \textbf{Caso del límite infinito:} Si asumimos que la función es continua en el toro ($f \in C(\mathbb{T})$) y por tanto acotada, su norma suprema domina a todas las demás normas $p$:
    \begin{equation*}
        \left( \frac{1}{2\pi} \int_{-\pi}^{\pi} |f(t)|^p dt \right)^{1/p} \le \left( \|f\|_\infty^p \frac{1}{2\pi} \int_{-\pi}^{\pi} dt \right)^{1/p} = \|f\|_\infty
    \end{equation*}
    
    \item \textbf{Densidad:} Las funciones continuas $C(\mathbb{T})$ forman un conjunto denso en $L_1(\mathbb{T})$ (y, en general, en cualquier $L_p(\mathbb{T})$). Dicho de otra manera: dada una $f \in L_1$ y cualquier margen de error $\varepsilon > 0$, siempre podremos encontrar una función continua $g \in C(\mathbb{T})$ tal que $\|f - g\|_1 < \varepsilon$. 
    
    Como conclusión general de este apartado, la jerarquía de densidades es $\mathcal{P}(\mathbb{T}) \subseteq C(\mathbb{T}) \subseteq L_1(\mathbb{T})$.
\end{enumerate}
\end{proposicion}

\begin{proof}
\textbf{(1):} Expresamos la integral de $|f|^p$ como el producto $|f|^p \cdot 1$. Aplicamos la desigualdad de Hölder con el exponente $r = q/p$ y su respectivo conjugado $r'$:
\begin{align*}
    \frac{1}{2\pi} \int_{-\pi}^{\pi} |f(t)|^p dt &= \frac{1}{2\pi} \int_{-\pi}^{\pi} |f(t)|^p \cdot 1 dt \\
    &\le \left( \frac{1}{2\pi} \int_{-\pi}^{\pi} \left(|f(t)|^p\right)^r dt \right)^{1/r} \cdot \left( \frac{1}{2\pi} \int_{-\pi}^{\pi} 1^{r'} dt \right)^{1/r'} \\
    &= \left( \frac{1}{2\pi} \int_{-\pi}^{\pi} |f(t)|^q dt \right)^{p/q} \cdot 1
\end{align*}
Elevando ambos lados a la potencia $1/p$, llegamos a la acotación directa $\|f\|_p \le \|f\|_q$.

\medskip
\textbf{(3):}
\begin{itemize}
    \item \textbf{Caso 1:} Si $f$ es una función indicadora estricta $f = \chi_{[a,b]}$, construimos una función continua $g_\delta(t)$ que aproxima a $f$ mediante un perfil trapezoidal. Definimos $g_\delta(t)$ explícitamente fijando un margen $\delta > 0$ en los extremos del intervalo:
    \begin{equation*}
        g_\delta(t) = \begin{cases} 
        0, & t \le a-\delta \\
        \frac{t - (a-\delta)}{\delta}, & a-\delta < t < a \\
        1, & a \le t \le b \\
        \frac{b+\delta - t}{\delta}, & b < t < b+\delta \\
        0, & t \ge b+\delta 
        \end{cases}
    \end{equation*}
    
    A continuación, evaluamos la norma $L_1$ de la discrepancia entre ambas funciones. Dado que $f(t) = g_\delta(t)$ tanto dentro del intervalo cerrado $[a,b]$ como fuera de $[a-\delta, b+\delta]$, el error se concentra exclusivamente en las dos franjas de transición (los triángulos sombreados):
    \begin{align*}
        \|f - g_\delta\|_1 &= \frac{1}{2\pi} \int_{-\pi}^{\pi} |f(t) - g_\delta(t)| dt \\
        &= \frac{1}{2\pi} \left( \int_{a-\delta}^{a} |0 - g_\delta(t)| dt + \int_{b}^{b+\delta} |0 - g_\delta(t)| dt \right) \\
        &= \frac{1}{2\pi} \left( \int_{a-\delta}^{a} \frac{t - a + \delta}{\delta} dt + \int_{b}^{b+\delta} \frac{b + \delta - t}{\delta} dt \right)
    \end{align*}
    Cada una de estas integrales representa el área de un triángulo de base $\delta$ y altura $1$. Por lo tanto, el área de cada uno es $\frac{\delta}{2}$. Sumando ambas contribuciones:
    \begin{equation*}
        \|f - g_\delta\|_1 = \frac{1}{2\pi} \left( \frac{\delta}{2} + \frac{\delta}{2} \right) = \frac{\delta}{2\pi}
    \end{equation*}
    Para garantizar que la norma del error sea estrictamente menor que un $\varepsilon$ arbitrario ($\|f - g_\delta\|_1 < \varepsilon$), basta con escoger el parámetro de transición como $\delta < 2\pi\varepsilon$.

    \item \textbf{Caso 2:} Si $f$ es una función simple, es decir, una combinación lineal finita de funciones características de intervalos disjuntos $f = \sum_{j=1}^n c_j \chi_{[a_j, b_j]}$, podemos aplicar el resultado del Caso 1 a cada uno de los términos. Dado $\varepsilon > 0$, construimos para cada $j$ una función continua $g_{\delta,j}$ tal que $\|\chi_{[a_j, b_j]} - g_{\delta,j}\|_1 < \frac{\varepsilon}{n|c_j|}$. Definiendo $g = \sum_{j=1}^n c_j g_{\delta,j}$, la función $g$ será continua y:
    \begin{equation*}
        \|f - g\|_1 = \left\| \sum_{j=1}^n c_j \chi_{[a_j, b_j]} - \sum_{j=1}^n c_j g_{\delta,j} \right\|_1 \le \sum_{j=1}^n |c_j| \left\| \chi_{[a_j, b_j]} - g_{\delta,j} \right\|_1 < \sum_{j=1}^n |c_j| \frac{\varepsilon}{n|c_j|} = \varepsilon
    \end{equation*}

    \item \textbf{Caso 3:} Si $f \in L_1(\mathbb{T})$ es cualquier función integrable, sabemos (por resultados clásicos de teoría de la medida o de la integral de Riemann) que siempre se puede aproximar por una función simple $s$ tal que $\|f - s\|_1 < \varepsilon/2$. Por el Caso 2, dado que $s$ es una función simple, podemos encontrar una función continua $g$ tal que $\|s - g\|_1 < \varepsilon/2$. Ahora, aplicando la desigualdad triangular:
    \begin{equation*}
        \|f - g\|_1 = \|f - s + s - g\|_1 \le \|f - s\|_1 + \|s - g\|_1 < \frac{\varepsilon}{2} + \frac{\varepsilon}{2} = \varepsilon
    \end{equation*}
    Con esto demostramos que para cualquier función $f \in L_1$, existe una función continua $g$ que se encuentra a una distancia menor que $\varepsilon$ en la norma $L_1$, lo que concluye la demostración de la densidad.
\end{itemize}
\end{proof}

\begin{observacion}{Jerarquía de espacios funcionales}{obs_jerarquia_espacios}
    Para el análisis de Fourier, se tiene la siguiente cadena de inclusiones densas en el toro $\mathbb{T}$:
    \[
    \mathcal{P}(\mathbb{T}) \subset \mathcal{C}(\mathbb{T}) \subset L_1(\mathbb{T})
    \]
    Es decir, los polinomios trigonométricos $\mathcal{P}(\mathbb{T})$ son densos en el espacio de funciones continuas $\mathcal{C}(\mathbb{T})$, el cual a su vez es denso en el espacio de funciones integrables $L_1(\mathbb{T})$.
\end{observacion}

\begin{proposicion}{Coeficientes de un polinomio trigonométrico}{coef_pol_trig}
Dado un polinomio trigonométrico $P(t) = \sum_{k=-M}^M c_k e^{ikt}$, sus coeficientes de Fourier $\hat{P}(n)$ coinciden de forma exacta con los propios coeficientes $c_n$.

\textit{Demostración:} Al tratarse de una suma finita, la linealidad de la integral nos permite intercambiar sin problemas el sumatorio y la integración:
\begin{equation*}
    \hat{P}(n) = \frac{1}{2\pi} \int_{-\pi}^{\pi} \left( \sum_{k=-M}^M c_k e^{ikt} \right) e^{-int} dt = \sum_{k=-M}^M c_k \left( \frac{1}{2\pi} \int_{-\pi}^{\pi} e^{i(k-n)t} dt \right)
\end{equation*}
Por la ortogonalidad natural del sistema, la integral vale $1$ si $k=n$ y $0$ en el resto de casos, colapsando toda la suma a:
\begin{equation*}
    \hat{P}(n) = \begin{cases} c_n & \text{si } -M \le n \le M \\ 0 & \text{en cualquier otro caso} \end{cases}
\end{equation*}
\end{proposicion}

    a_n &= \hat{f}(n) + \hat{f}(-n) \\
    b_n &= i(\hat{f}(n) - \hat{f}(-n))
\end{align*}
\end{observacion}

\begin{definicion}{El Coeficiente y la Serie de Fourier}{def_fourier_formal}
Para toda función integrable $f \in L_1(\mathbb{T})$, definimos el \textbf{coeficiente de Fourier $n$-ésimo} (con $n \in \mathbb{Z}$) de la siguiente manera:
\begin{equation*}
    \hat{f}(n) = \frac{1}{2\pi} \int_{-\pi}^{\pi} f(t) e^{-int} dt
\end{equation*}
Esta operación está siempre bien definida porque la acotación modular $|f(t)e^{-int}| = |f(t)|$ garantiza que el integrando permanece dentro de las fronteras de $L_1$, produciendo que $\hat{f}(n) \in \mathbb{C}$ exista de manera finita.

Con la colección de estos coeficientes generamos la \textbf{Serie de Fourier} asociada a $f$:
\begin{equation*}
    f(t) \sim \sum_{n=-\infty}^{\infty} \hat{f}(n) e^{int}
\end{equation*}
Para poder manejar y analizar esta serie infinita definimos la \textbf{suma parcial n-ésima}, $S(f,n)(t) = \sum_{k=-n}^{n} \hat{f}(k) e^{ikt}$, que resulta ser un polinomio trigonométrico que mapea $[-\pi,\pi] \to \mathbb{C}$.
\end{definicion}

\begin{observacion}{Dudas sobre la convergencia}{conv_uniforme_fourier}
\textbf{Pregunta clave:} ¿Podemos asumir que la suma parcial $S(f,n)$ acabará convergiendo (y además uniformemente) de regreso hacia nuestra $f$ original? 

La respuesta es un rotundo \textbf{no}. La razón profunda es topológica: si el límite uniforme de los polinomios trigonométricos $S(f,n)$ existiese siempre, forzosamente daría a luz una función continua. Sin embargo, el espacio $L_1(\mathbb{T})$ está completamente repleto de funciones discontinuas, por lo que la convergencia uniforme general es imposible de garantizar sin hipótesis de regularidad añadidas.
\end{observacion}

\begin{ejemplo}{Ortogonalidad de la base exponencial}{ejemplo_base_exp}
Para ver cómo funciona la maquinaria sobre sus propios cimientos, analicemos la función base $\phi_m(t) = e^{imt}$. Su coeficiente de Fourier $\hat{\phi}_m(n)$ evaluará la presencia de la frecuencia $n$ en la frecuencia $m$:
\begin{equation*}
    \hat{\phi}_m(n) = \frac{1}{2\pi} \int_{-\pi}^{\pi} e^{imt} e^{-int} dt = \frac{1}{2\pi} \int_{-\pi}^{\pi} e^{i(m-n)t} dt
\end{equation*}

\begin{itemize}
    \item \textbf{Si $m = n$:} la integral colapsa a la unidad básica $\frac{1}{2\pi} \int_{-\pi}^{\pi} e^0 dt = 1$.
    \item \textbf{Si $m \ne n$:} llamando a su diferencia no nula $k = m-n$, aplicamos la primitiva:
    \begin{equation*}
        \hat{\phi}_m(n) = \frac{1}{2\pi i k} \left[ e^{ikt} \right]_{-\pi}^{\pi} = \frac{1}{2\pi i k} \left( e^{ik\pi} - e^{-ik\pi} \right) = 0
    \end{equation*}
\end{itemize}

El resultado es la clásica Delta de Kronecker, demostrando que las funciones armónicas filtran sus frecuencias de manera ortonormal:
\begin{equation*}
    \hat{\phi}_m(n) = \begin{cases} 
        1 & \text{si } n=m \\ 
        0 & \text{si } n \ne m 
    \end{cases}
\end{equation*}
\end{ejemplo}

Para $n = 0$:
\[
\hat{f}(0) = \frac{1}{2} \int_{-1}^{1} t^2 \, dt = \frac{1}{2} \left[ \frac{t^3}{3} \right]_{-1}^{1} = \frac{1}{3}.
\]

Para $n \neq 0$, utilizando integración por partes con $a = i n \pi$ (donde $a^2 = -n^2\pi^2$ y $a^3 = -i n^3 \pi^3$):
\begin{align*}
\hat{f}(n) &= \frac{1}{2} \int_{-1}^{1} t^2 e^{-i n \pi t} \, dt = \frac{1}{2} \left[ \frac{a^2 t^2 + 2at + 2}{-a^3} e^{-a t} \right]_{-1}^{1} \\
&= \frac{1}{2 i n^3 \pi^3} \left[ (-n^2\pi^2 + 2\pi i n + 2) e^{-i n \pi} - (-n^2\pi^2 - 2\pi i n + 2) e^{i n \pi} \right].
\end{align*}

Distinguiendo por paridad:
\begin{itemize}
    \item $n$ par ($n = 2k$): $e^{\pm(2k)\pi i} = 1$.
    \begin{align*}
        \hat{f}(n) &= \frac{1}{2 i (2k)^3 \pi^3} \Big[ \left(-(2k)^2\pi^2 + 2\pi i (2k) + 2\right) \\
        &\quad - \left(-(2k)^2\pi^2 - 2\pi i (2k) + 2\right) \Big] \\
        &= \frac{1}{2 i (2k)^3 \pi^3} \big[ 4\pi i (2k) \big] = \frac{2}{(2k)^2 \pi^2}.
    \end{align*}
    
    \item $n$ impar ($n = 2k+1$): $e^{\pm(2k+1)\pi i} = -1$.
    \begin{align*}
        \hat{f}(n) &= \frac{1}{2 i (2k+1)^3 \pi^3} \Big[ \left(-(2k+1)^2\pi^2 + 2\pi i (2k+1) + 2\right)(-1) \\

        &= \frac{1}{2 i (2k+1)^3 \pi^3} \big[ -4\pi i (2k+1) \big] = -\frac{2}{(2k+1)^2 \pi^2}.
    \end{align*}
\end{itemize}

En general, usando $e^{-i n \pi} = (-1)^n$, obtenemos $\hat{f}(n) = \frac{2(-1)^n}{n^2 \pi^2}$.
La serie compleja resulta:

f(t) \sim \frac{1}{3} + \sum_{\substack{n \in \mathbb{Z} \\ n \neq 0}} \frac{2(-1)^n}{n^2 \pi^2} e^{i n \pi t}.

\begin{observacion}{Convergencia de las sumas parciales $S_n(f)$}{obs_convergencia}
\begin{enumerate}
    \item [(1)]$f \in L_1$: ¿$\|f - S_n(f)\|_1 \to 0$? ¿$S_n(f) \to f$ en $L_1$? $\implies$ \textbf{NO} (Kolmogorov dio un ejemplo).
    \item[(2)] $f \in L_1$: ¿$\|f - S_n(f)\|_1 \to 0$? ¿$S_n(f) \to f$ en $L_1$? $\implies$ \textbf{NO} (Shilov dio un ejemplo).
    \item[(3)] $f \in L_2$: ¿$\|f - S_n(f)\|_2 \to 0$? ¿$S_n(f) \to f$ en $L_2$? $\implies$ \textbf{SÍ} (veremos).
    \item[(4)] $f \in C(\mathbb{T})$: ¿$\lim_n \|f - S_n(f)\|_\infty = 0$ (Convergencia uniforme) o al menos $\lim_n |f(t) - S_n(f, t)| = 0, \; \forall t$? $\implies$ \textbf{NO} (Du Bois Reymond ejemplo (1873) func continua cuya suma parcial no converge en $t = 0$).
    \item[(5)] Si $f \in C(\mathbb{T})$, ¿podemos recuperar $f$ a partir de sus coeficientes? $\implies$ \textbf{SÍ} Fejer (19 años).
\end{enumerate}
\end{observacion}

\begin{proposicion}{Propiedad 1.4}{prop_1_4}
Sea $f \in L_1(\mathbb{T})$. Se cumplen las siguientes propiedades:
\begin{enumerate}
    \item[(1)] \textbf{Acotación:} $|\hat{f}(n)| \le \|f\|_1$ para todo $n \in \mathbb{Z}$.
    \item[(2)] \textbf{Lema de Riemann-Lebesgue:} Si tomo $\varepsilon > 0$, existe $n_0 \in \mathbb{N}$ tal que $|n| \ge n_0 \implies |\hat{f}(n)| < \varepsilon$.
    \item[(3)] \textbf{Unicidad nula:} Si $\hat{f}(n) = 0$ para todo $n \in \mathbb{Z}$, entonces $f = 0$ casi por todas partes (a.e.). Creo, no lo he copiado
    \item[(4)] \textbf{Unicidad general:} Dados $f, g \in L_1$, si $\hat{f}(n) = \hat{g}(n)$ para todo $n \in \mathbb{Z}$, entonces $f = g$ casi por todas partes (a.e.).
\end{enumerate}
\end{proposicion}

\begin{proof}[Demostración de (1)]
Aplicando el valor absoluto a la definición del coeficiente de Fourier y usando la desigualdad triangular para integrales:
\[
|\hat{f}(n)| = \left| \frac{1}{2\pi} \int_{-\pi}^\pi f(t) e^{-i n t} \, dt \right| \le \frac{1}{2\pi} \int_{-\pi}^\pi |f(t)| \cdot \underbrace{|e^{-i n t}|}_{1} \, dt \le \|f\|_1.
\]
\end{proof}

\begin{proof}[Demostración de (2)]
Sea $f \in L_1$ y tomo $\varepsilon > 0$. Quiero encontrar un $n_0$ tal que $|n| \ge n_0 \implies |\hat{f}(n)| < \varepsilon$. 

Sabemos que el conjunto de los polinomios trigonométricos $\mathcal{P}(\mathbb{T})$ es denso en $L_1$, lo que implica que $\exists P \in \mathcal{P}(\mathbb{T})$ con $\|f - P\|_1 < \varepsilon$, donde $P(t) = \sum_{n=-N}^N c_n e^{i n t}$. 

Ahora, sumando y restando $\hat{P}(n)$:
\[
|\hat{f}(n)| = |\hat{f}(n) - \hat{P}(n) + \hat{P}(n)| \le |\hat{f}(n) - \hat{P}(n)| + |\hat{P}(n)| = |\widehat{f - P}(n)| + |\hat{P}(n)|
\]
\[
\le \|f - P\|_1 + |\hat{P}(n)| < \varepsilon + |\hat{P}(n)|, \quad \forall n \in \mathbb{Z}.
\]
Recordemos que los coeficientes del polinomio trigonométrico son:
\[
\hat{P}(n) = \begin{cases} c_n & \text{si } -N \le n \le N \\ 0 & \text{en otro caso} \end{cases}
\]
Si $|n| > max\{M, N\}$ valdría con N? $ \implies \hat{P}(n) = 0$. (Nota: asumiendo $max(M,N)$ como el grado máximo del polinomio). 

Tomando $n_0 = max\{M, N\} + 1$, si $|n| \ge n_0 \implies |\hat{f}(n)| < \varepsilon + 0 = \varepsilon$.
\end{proof}

\begin{proof}[Demostración de (3)]
Más adelante en el curso.
\end{proof}

\begin{proof}[Demostración de (4)]
Dados $f, g \in L_1$, defino una nueva función $h = f - g$. Por la linealidad de los coeficientes:
\[
\hat{h}(n) = \widehat{f - g}(n) = \hat{f}(n) - \hat{g}(n) = 0 \quad \forall n \in \mathbb{Z}.
\]
Aplicando la propiedad (3) a la función $h$, obtenemos directamente que:
\[
\hat{h}(n) = 0 \implies h = 0 \text{ a.e.} \implies f = g \text{ a.e.}
\]
\end{proof}

\begin{nota}{Límite de coeficientes}{nota_cvg}
Si $f_n \to f$ en $L_1$ ($\|f_n - f\|_1 \to 0$), entonces para cada $k \in \mathbb{Z}$:
\[
|\hat{f}_n(k) - \hat{f}(k)| = |\widehat{f_n - f}(k)| \le \|f_n - f\|_1 \to 0.
\]
Es decir, $|\hat{f}_n(k) - \hat{f}(k)| \to 0 \implies \hat{f}_n(k) \to \hat{f}(k)$. \\
Para cada $n$ tengo $f_n$. Para cada $k \in \mathbb{Z}$, $(\hat{f}_n(k))_n \subseteq \mathbb{C}$.
\end{nota}

\subsection{Convolución de Funciones}

\begin{definicion}{Convolución}{def_convolucion}
Tengo $f, g \in C(\mathbb{T})$. Para cada $t \in \mathbb{R}$ fijo considero $s \mapsto g(t - s) = g \circ \varphi$ (cont), donde $\varphi: \mathbb{R} \to \mathbb{R}$, $s \mapsto t - s$ (cont).
\[
\frac{1}{2\pi} \int_{-\pi}^\pi f(s) g(t - s) \, ds = (f * g)(t).
\]
\end{definicion}

\begin{proposicion}{Prop 1.5: Convolución en $L_1$}{prop_1_5}
Si $f, g \in L_1$, esta bien definida la convolución \\
\textbf{Idea:} $F(t, s) = f(s) g(t - s)$, $h(t) = \frac{1}{2\pi} \int F(t, s) \, ds$ bien def para casi todo $t \in [-\pi, \pi]$. Además: $h \in L_1(\mathbb{T})$ y $\|h\|_1 \le \|f\|_1 \|g\|_1$.
\end{proposicion}

\begin{proof}
Consideran las func:
\[
\varphi: [-\pi, \pi] \times [-\pi, \pi] \longrightarrow \mathbb{C}, \quad (t, s) \longmapsto e^{(t - s)i}
\]
\[
\psi: [-\pi, \pi] \times [-\pi, \pi] \longrightarrow \mathbb{C}, \quad (t, s) \longmapsto e^{-si}
\]
Para $f, g \in L_1$, identificamos $f(x) = f(e^{i x})$. \\
$g(t - s) = (g \circ \varphi)(t, s)$ y $f(s) = (f \circ \psi)(t, s)$ medibles porque composición de integrable con continua. \\
Defino $F: [-\pi, \pi] \times [-\pi, \pi] \longrightarrow \mathbb{C}$ dada por $(t, s) \longmapsto F(t, s) = f(s) g(t - s)$ medible.

Quiero aplicar Hobson-Tonelli. \\
Antes observamos que para cada $t$ fijo:
\[
\frac{1}{2\pi} \int_{-\pi}^\pi |g(t - s)| \, ds \underset{u = t - s}{=} \frac{1}{2\pi} \int_{-\pi - t}^{\pi - t} |g(u)| \, du = \frac{1}{2\pi} \int_{-\pi}^\pi |g(u)| \, du = \|g\|_1,
\]
ya que si $g$ es $2\pi$-periódica y $I_1, I_2$ intervalos long $2\pi \implies \int_{I_1} g(u) \, du = \int_{I_2} g(u) \, du$.

Integral doble:
\[
\int_{-\pi}^\pi \left( \int_{-\pi}^\pi |F(t, s)| \, dt \right) ds = \int_{-\pi}^\pi |f(s)| \left( \int_{-\pi}^\pi |g(t - s)| \, dt \right) ds = \int_{-\pi}^\pi |f(s)| \int_{-\pi}^\pi |g(u)| \, du < \infty 
\]
($f \in L_1 \quad, g \in L_1$). 


Por Hobson-Tonelli: $F$ integrable Lebesgue. \\
Fubini: Para casi todo $t \in [-\pi, \pi]$
\begin{itemize}
    \item $s \longmapsto F(t, s)$ integrable Lebesgue y $\hat{h}(t) = \int_{-\pi}^\pi F(t, s) \, ds$ es integrable Lebesgue.
\end{itemize}
Entonces $h(t) = \frac{1}{2\pi} \int_{-\pi}^\pi f(s) g(t - s) \, ds$ int Leb.

Además:
\[
\|h\|_1 = \frac{1}{2\pi} \int_{-\pi}^\pi |h(t)| \, dt = \frac{1}{2\pi} \int_{-\pi}^\pi \left| \frac{1}{2\pi} \int_{-\pi}^\pi f(s) g(t - s) \, ds \right| dt \le \frac{1}{2\pi} \cdot \frac{1}{2\pi} \int_{-\pi}^\pi \int_{-\pi}^\pi |f(s)| \cdot |g(t - s)| \, ds \, dt
\]
\[
= \frac{1}{2\pi} \cdot \frac{1}{2\pi} \int_{-\pi}^\pi \int_{-\pi}^\pi |f(s)| \cdot |g(t - s)| \, dt \, ds = \|f\|_1 \|g\|_1.
\]
\end{proof}

\subsection{Convolución y Núcleo de Dirichlet}

\begin{definicion}{Convolución en $L_1(\mathbb{T})$}{conv_l1}
Sean $f, g \in L_1(\mathbb{T})$. Se define la convolución de $f$ y $g$ como la función dada por:
\[
(f * g)(t) = \frac{1}{2\pi} \int_{-\pi}^\pi f(s) g(t - s) \, ds.
\]
Se cumple además la desigualdad:
\[
\|f * g\|_1 \le \|f\|_1 \|g\|_1.
\]
\end{definicion}

\subsubsection*{Ejemplos de Convolución}

\begin{ejemplo}{Exponenciales complejas}{exp_complejas}
Para cada $n \in \mathbb{Z}$, definimos $\phi_n(t) = e^{int}$. Calculando la convolución de dos exponenciales complejas:
\begin{align*}
(\phi_n * \phi_m)(t) &= \frac{1}{2\pi} \int_{-\pi}^\pi \phi_n(s) \phi_m(t - s) \, ds = \frac{1}{2\pi} \int_{-\pi}^\pi e^{ins} e^{im(t - s)} \, ds \\
&= \frac{1}{2\pi} e^{imt} \int_{-\pi}^\pi e^{i(n - m)s} \, ds \\
&= \frac{1}{2\pi} \cdot 2\pi \delta_{nm} e^{imt} = \delta_{nm} e^{imt}.
\end{align*}
Por tanto:
\[
\phi_n * \phi_m (t) = \delta_{nm} \phi_m(t).
\]
\end{ejemplo}

\begin{ejemplo}{Función característica epsicircular}{caract_epsicircular}
Sea $0 < \varepsilon < \pi$ y consideremos la función $\rho_\varepsilon = \frac{1}{2\varepsilon} \chi_{[-\varepsilon, \varepsilon]}$. Para $g \in L_1(\mathbb{T})$:
\begin{align*}
(\rho_\varepsilon * g)(t) &= \frac{1}{2\pi} \int_{-\pi}^\pi \rho_\varepsilon(s) g(t - s) \, ds = \frac{1}{2\pi} \int_{-\varepsilon}^\varepsilon \frac{1}{2\varepsilon} g(t - s) \, ds \\
&= \frac{1}{4\pi\varepsilon} \int_{-\varepsilon}^\varepsilon g(t - s) \, ds.
\end{align*}
Efectuando el cambio de variable $u = t - s$ ($du = -ds$, con $s = -\varepsilon \mapsto u = t + \varepsilon$ y $s = \varepsilon \mapsto u = t - \varepsilon$):
\[
(\rho_\varepsilon * g)(t) = -\frac{1}{4\pi\varepsilon} \int_{t+\varepsilon}^{t-\varepsilon} g(u) \, du = \frac{1}{4\pi\varepsilon} \int_{t-\varepsilon}^{t+\varepsilon} g(u) \, du.
\]
\end{ejemplo}

\begin{observacion}{Unidad en el álgebra de convolución}{unidad_conv}
\textbf{Pregunta:} En $(L_1(\mathbb{T}), *)$, ¿existe elemento neutro (unidad)? Es decir, ¿existe $f \in L_1(\mathbb{T})$ tal que $f * g = g$ para toda $g \in L_1(\mathbb{T})$?

\textbf{Respuesta:} \textbf{No}. Se puede demostrar como consecuencia de la \cref{prop:fourier_conv} (ver ejercicio a continuación).
\end{observacion}

\begin{proposicion}{Coeficientes de Fourier de una convolución}{fourier_conv}
Sean $f, g \in L_1(\mathbb{T})$ y $n \in \mathbb{Z}$. Entonces:
\[
\widehat{(f * g)}(n) = \hat{f}(n) \cdot \hat{g}(n).
\]
\end{proposicion}

\begin{proof}
Para $n \in \mathbb{Z}$, por definición de los coeficientes de Fourier:
\begin{align*}
\widehat{(f * g)}(n) &= \frac{1}{2\pi} \int_{-\pi}^\pi (f * g)(t) e^{-int} \, dt \\
&= \frac{1}{2\pi} \int_{-\pi}^\pi \left[ \frac{1}{2\pi} \int_{-\pi}^\pi f(s) g(t - s) \, ds \right] e^{-int} \, dt.
\end{align*}
Puesto que $e^{-int} = e^{-ins} e^{-in(t - s)}$ no depende de $s$ de forma directa respecto a la medida completa, por el teorema de Fubini se intercambia el orden de integración:
\begin{align*}
\widehat{(f * g)}(n) &= \frac{1}{2\pi} \int_{-\pi}^\pi \frac{1}{2\pi} \left[ \int_{-\pi}^\pi f(s) e^{-ins} g(t - s) e^{-in(t - s)} \, dt \right] ds \\
&= \left(\frac{1}{2\pi}\right)^2 \int_{-\pi}^\pi f(s) e^{-ins} \left[ \int_{-\pi}^\pi g(t - s) e^{-in(t - s)} \, dt \right] ds.
\end{align*}
Para cada $s$ fijo, efectuando el cambio de variable $u = t - s$ y utilizando la $2\pi$-periodicidad de las funciones:
\[
\int_{-\pi}^\pi g(t - s) e^{-in(t - s)} \, dt = \int_{-\pi}^\pi g(u) e^{-inu} \, du = 2\pi \hat{g}(n).
\]
Sustituyendo en la expresión anterior:
\begin{align*}
\widehat{(f * g)}(n) &= \left(\frac{1}{2\pi}\right)^2 \int_{-\pi}^\pi f(s) e^{-ins} \cdot 2\pi \hat{g}(n) \, ds \\
&= \left( \frac{1}{2\pi} \int_{-\pi}^\pi f(s) e^{-ins} \, ds \right) \hat{g}(n) = \hat{f}(n) \cdot \hat{g}(n).
\end{align*}
\end{proof}

\begin{nota}{Ejercicio sobre periodicidad de integrales}{ej_per_int}
Demostrar que si $h\colon \mathbb{R} \to \mathbb{C}$ es $2\pi$-periódica e $I_1, I_2$ son intervalos de longitud $2\pi$, entonces:
\[
\int_{I_1} h(t) \, dt = \int_{I_2} h(t) \, dt.
\]
\end{nota}

\subsection{Convergencia de la Serie de Fourier y Núcleo de Dirichlet}

Estudio de la convergencia de $S_N(f, t)$:
\begin{align*}
S_N(f, t) &= \sum_{k=-N}^N \hat{f}(k) e^{ikt} = \sum_{k=-N}^N \left( \frac{1}{2\pi} \int_{-\pi}^\pi f(s) e^{-iks} \, ds \right) e^{ikt} \\
&= \frac{1}{2\pi} \int_{-\pi}^\pi f(s) \sum_{k=-N}^N e^{-iks} e^{ikt} \, ds \\
&= \frac{1}{2\pi} \int_{-\pi}^\pi f(s) \left( \sum_{k=-N}^N e^{ik(t - s)} \right) ds.
\end{align*}

\begin{definicion}{Núcleo de Dirichlet}{nucleo_dirichlet}
Para cada $N \in \mathbb{N}$, se define el \textbf{núcleo de Dirichlet $N$-ésimo} como:
\[
D_N(t) = \sum_{k=-N}^N e^{ikt}.
\]
\end{definicion}

La suma parcial se expresa como una convolución:
\[
S_N(f, t) = \frac{1}{2\pi} \int_{-\pi}^\pi f(s) D_N(t - s) \, ds = (f * D_N)(t).
\]

\begin{nota}{Notación asintótica}{notacion_asintotica}
Si $(a_n)_n, (b_n)_n \subseteq \mathbb{C}$, se define:
\[
(a_n) \sim (b_n) \iff \exists A, B > 0 \text{ tales que } A a_n \le b_n \le B a_n.
\]
\end{nota}

\begin{proposicion}{Propiedades del núcleo de Dirichlet}{prop_dirichlet}
\begin{enumerate}
    \item $S_N(f, t) = (f * D_N)(t)$ para toda $f \in L_1(\mathbb{T})$.
    \item Expresión cerrada:
    \[
    D_N(t) = \begin{cases}
    2N + 1, & \text{si } t = 2k\pi, \; k \in \mathbb{Z}, \\[1ex]
    \dfrac{e^{i\left(N + \frac{1}{2}\right)t} - e^{-i\left(N + \frac{1}{2}\right)t}}{e^{i t/2} - e^{-i t/2}} = \dfrac{\operatorname{sen}\left(\left(N + \frac{1}{2}\right)t\right)}{\operatorname{sen}(t/2)}, & \text{en otro caso.}
    \end{cases}
    \]
    \item $D_N(t)$ es continua, $2\pi$-periódica y par ($D_N(t) = D_N(-t)$).
    \item Masa total unitaria:
    \[
    \frac{1}{2\pi} \int_{-\pi}^\pi D_N(t) \, dt = 1.
    \]
    \item $\|D_N\|_1 \sim \log N$.
    \item $\|D_N\|_\infty = 2N + 1$.
\end{enumerate}
\end{proposicion}

\begin{proof}
\textbf{Demostración de (1):} Directa a partir de la inversión de sumatorio e integral mostrada anteriormente.

\medskip
\textbf{Demostración de (2):}
\begin{itemize}
    \item Si $t = 2n\pi$ para $n \in \mathbb{Z}$:
    \[
    e^{ikt} = e^{ik(2n\pi)} = 1 \implies D_N(t) = \sum_{k=-N}^N 1 = 2N + 1.
    \]
    \item Si $t \neq 2k\pi$:
    \begin{align*}
    D_N(t) &= \sum_{k=-N}^N e^{ikt} = e^{-iNt} \sum_{u=0}^{2N} (e^{it})^u = e^{-iNt} \frac{(e^{it})^{2N+1} - 1}{e^{it} - 1} \\
    &= \frac{e^{i(N+1)t} - e^{-iNt}}{e^{it} - 1} = \frac{e^{-it/2}}{e^{-it/2}} \cdot \frac{e^{i(N+1)t} - e^{-iNt}}{e^{it} - 1} \\
    &= \frac{e^{i\left(N+\frac{1}{2}\right)t} - e^{-i\left(N+\frac{1}{2}\right)t}}{e^{it/2} - e^{-it/2}} = \frac{\operatorname{sen}\left(\left(N + \frac{1}{2}\right)t\right)}{\operatorname{sen}(t/2)}.
    \end{align*}
\end{itemize}

\medskip
\textbf{Demostración de (3):} Se deduce inmediatamente de la expresión trigonométrica en (2).

\medskip
\textbf{Demostración de (4):}
Intercambiando integral y sumatorio:
\[
\frac{1}{2\pi} \int_{-\pi}^\pi D_N(t) \, dt = \frac{1}{2\pi} \sum_{k=-N}^N \int_{-\pi}^\pi e^{ikt} \, dt.
\]
Para $k \neq 0$, $\int_{-\pi}^\pi e^{ikt} \, dt = 0$; para $k = 0$, $\int_{-\pi}^\pi 1 \, dt = 2\pi$. Por tanto:
\[
\frac{1}{2\pi} \int_{-\pi}^\pi D_N(t) \, dt = \frac{1}{2\pi} (2\pi) = 1.
\]
\textbf{Demostración de (5):}
Por paridad del núcleo de Dirichlet ($D_N(-t) = D_N(t)$), calculamos su norma $L_1$:


$$\begin{aligned}\Vert{}D_N\Vert{}_1 &= \frac{1}{2\pi} \int_{-\pi}^\pi \vert{}D_N(t)\vert{} \, dt = \frac{2}{2\pi} \int_0^\pi \vert{}D_N(t)\vert{} \, dt \\ &= \frac{1}{\pi} \int_0^\pi \left\vert{} \frac{\operatorname{sen}\left(\left(N + \frac{1}{2}\right)t\right)}{\operatorname{sen}(t/2)} \right\vert{} dt.\end{aligned}$$

Efectuando el cambio de variable $u = t/2$ (con lo que $dt = 2\,du$, y los límites pasan a ser $u \in [0, \pi/2]$):


$$\Vert{}D_N\Vert{}_1 = \frac{1}{\pi} \int_0^{\pi/2} \frac{\left\vert{}\operatorname{sen}\left((2N + 1)u\right)\right\vert{}}{\left\vert{}\operatorname{sen} u\right\vert{}} 2\,du = \frac{2}{\pi} \int_0^{\pi/2} \frac{\left\vert{}\operatorname{sen}\left((2N + 1)u\right)\right\vert{}}{\operatorname{sen} u} \, du$$

\medskip

\textbf{Paso 1: Equivalencia asintótica del denominador.}
Recordemos que:


$$\lim_{x \to 0} \frac{\operatorname{sen} x}{x} = 1$$

Fijando $\varepsilon = \frac{1}{2}$, por definición de límite existe $\delta > 0$ tal que si $0 < \vert{}x\vert{} < \delta$:


$$\left\vert{} \frac{\operatorname{sen} x}{x} - 1 \right\vert{} < \frac{1}{2} \implies \frac{1}{2} < \frac{\operatorname{sen} x}{x} < \frac{3}{2}$$

Para el resto del intervalo, si $0 < \delta \le x \le \pi/2$, la función $x \mapsto \frac{\operatorname{sen} x}{x}$ es continua en el compacto $[\delta, \pi/2]$, por lo que alcanza un mínimo y un máximo estrictamente positivos:


$$\exists\, M_1, M_2 > 0 \quad \text{tales que} \quad M_1 \le \frac{\operatorname{sen} x}{x} \le M_2, \quad \forall x \in [\delta, \pi/2]$$

Combinando ambas regiones, existen constantes globales $A, B > 0$ tales que:


$$A \le \frac{\operatorname{sen} x}{x} \le B, \quad \forall x \in (0, \pi/2] \implies Ax \le \operatorname{sen} x \le Bx$$

De aquí, invirtiendo la desigualdad se deduce:


$$\frac{1}{B u} \le \frac{1}{\operatorname{sen} u} \le \frac{1}{A u}, \quad \forall u \in (0, \pi/2]$$

Multiplicando por $\vert{}\operatorname{sen}((2N+1)u)\vert{} \ge 0$ e integrando en $[0, \pi/2]$:


$$\frac{1}{B} \int_0^{\pi/2} \frac{\vert{}\operatorname{sen}((2N+1)u)\vert{}}{u} \, du \le \int_0^{\pi/2} \frac{\vert{}\operatorname{sen}((2N+1)u)\vert{}}{\operatorname{sen} u} \, du \le \frac{1}{A} \int_0^{\pi/2} \frac{\vert{}\operatorname{sen}((2N+1)u)\vert{}}{u} \, du$$

Por definición de equivalencia asintótica:


$$\frac{2}{\pi} \int_0^{\pi/2} \frac{\vert{}\operatorname{sen}((2N + 1)u)\vert{}}{\operatorname{sen} u} \, du \sim \frac{2}{\pi} \int_0^{\pi/2} \frac{\vert{}\operatorname{sen}((2N + 1)u)\vert{}}{u} \, du$$

\medskip
\textbf{Paso 2: Cambio de escala y descomposición en partición.}
Haciendo el cambio de variable $v = (2N + 1)u$, de donde $u = \frac{v}{2N+1}$ y $du = \frac{1}{2N + 1} \, dv$ (con $u = 0 \mapsto v = 0$ y $u = \pi/2 \mapsto v = (2N+1)\pi/2$):


$$\begin{aligned}\frac{2}{\pi} \int_0^{\pi/2} \frac{\vert{}\operatorname{sen}((2N + 1)u)\vert{}}{u} \, du &= \frac{2}{\pi} \int_0^{(2N+1)\pi/2} \frac{\vert{}\operatorname{sen} v\vert{}}{\frac{v}{2N + 1}} \frac{1}{2N + 1} \, dv \\ &= \frac{2}{\pi} \int_0^{(2N+1)\pi/2} \frac{\vert{}\operatorname{sen} v\vert{}}{v} \, dv\end{aligned}$$

Particionamos el intervalo de integración $[0, (2N+1)\pi/2]$ en submúltiplos de $\pi/2$, correspondientes a los intervalos $\left[ \frac{k\pi}{2}, \frac{(k+1)\pi}{2} \right]$ para $k = 0, 1, \dots, 2N$:


$$\frac{2}{\pi} \int_0^{(2N+1)\pi/2} \frac{\vert{}\operatorname{sen} v\vert{}}{v} \, dv = \frac{2}{\pi} \sum_{k=0}^{2N} \int_{k\pi/2}^{(k+1)\pi/2} \frac{\vert{}\operatorname{sen} v\vert{}}{v} \, dv$$

\medskip
\textbf{Paso 3: Acotación de $1/v$ en cada semionda.}
En cada subintervalo $\left[ \frac{k\pi}{2}, \frac{(k+1)\pi}{2} \right]$, puesto que $v > 0$, la función $v \mapsto \frac{1}{v}$ es decreciente:


$$\frac{k\pi}{2} \le v \le (k+1)\frac{\pi}{2} \implies \frac{2}{\pi} \frac{1}{k+1} \le \frac{1}{v} \le \frac{2}{\pi} \frac{1}{k}$$

Observemos que para $k \ge 1$:


$$\frac{k+1}{k} = 1 + \frac{1}{k} \le 1 + 1 = 2 \implies \frac{1}{k+1} \ge \frac{1}{2k}$$

Por lo tanto, la cota inferior se reduce a:


$$\frac{1}{v} \ge \frac{2}{\pi} \frac{1}{k+1} \ge \frac{1}{\pi} \frac{1}{k}, \quad \text{es decir,} \quad \frac{2}{\pi} \frac{1}{k} \ge \frac{1}{v} \ge \frac{1}{\pi} \frac{1}{k} \quad (\forall k \ge 1)$$

Esto establece que para cada $k \ge 1$, el factor $\frac{1}{v}$ es asintóticamente equivalente a $\frac{1}{k}$:


$$\frac{1}{v} \sim \frac{1}{k}$$

\medskip
\textbf{Paso 4: Integral de las semiondas trigonométricas y conclusión.}
En cada intervalo $\left[ \frac{k\pi}{2}, \frac{(k+1)\pi}{2} \right]$, la función $\operatorname{sen} v$ no cambia de signo, por lo que el valor absoluto de su integral coincide con la integral de una semionda:


$$\int_{k\pi/2}^{(k+1)\pi/2} \vert{}\operatorname{sen} v\vert{} \, dv = \left\vert{} \int_{k\pi/2}^{(k+1)\pi/2} \operatorname{sen} v \, dv \right\vert{} = 1$$

Sustituyendo esto en la suma (descartando el término $k=0$, que es una integral finita que solo aporta una constante acotada):


$$\begin{aligned}\frac{2}{\pi} \sum_{k=0}^{2N} \int_{k\pi/2}^{(k+1)\pi/2} \frac{\vert{}\operatorname{sen} v\vert{}}{v} \, dv &\sim \frac{2}{\pi} \sum_{k=1}^{2N} \frac{1}{k} \int_{k\pi/2}^{(k+1)\pi/2} \vert{}\operatorname{sen} v\vert{} \, dv \\ &= \frac{2}{\pi} \sum_{k=1}^{2N} \frac{1}{k}\end{aligned}$$

Para estudiar la convergencia, observamos la relación entre la integral y la suma:


$$\int_1^M \frac{1}{x} dx = \sum_{k=1}^{M-1} \int_k^{k+1} \frac{1}{x} dx \le \sum_{k=1}^{M-1} \int_k^{k+1} \frac{1}{k} dx = \sum_{k=1}^{M-1} \frac{1}{k} \le \sum_{k=1}^M \frac{1}{k}$$

Sabemos que $\int_1^M \frac{1}{x} dx = \log(M)$, ya que para $k \le x \le k+1$, se tiene $\frac{1}{k} \ge \frac{1}{x} \ge \frac{1}{k+1}$. Queremos ver que $\sum_{k=1}^M \frac{1}{k} \sim \log M$.
Por otro lado:


$$\log(M) = \int_1^M \frac{1}{x} dx \ge \frac{1}{2} \sum_{k=1}^{M-1} \frac{1}{k} = \frac{1}{2} \left( \sum_{k=1}^M \frac{1}{k} - \frac{1}{M} \right) \le \frac{1}{2} \left( \sum_{k=1}^M \frac{1}{k} - \frac{1}{2} \sum_{k=1}^M \frac{1}{k} \right)$$

Observemos que:


$$\frac{1}{M} \le \sum_{k=1}^{M-1} \frac{1}{k}, \quad \frac{1}{2M} \le \frac{1}{2} \sum_{k=1}^{M-1} \frac{1}{k}, \quad \frac{1}{M} \le \frac{1}{2} \sum_{k=1}^{M-1} \frac{1}{k} + \frac{1}{2M} = \frac{1}{2} \sum_{k=1}^M \frac{1}{k}$$


Luego $A_1 = \frac{1}{4} \sum_{k=1}^M \frac{1}{k}$.

Aplicando esto para $M=2N$, se concluye finalmente, por el comportamiento de la serie armónica, que:


$$\sum_{k=1}^{2N} \frac{1}{k} \sim \log(2N) = \log 2 + \log N \sim \log N$$

Concluimos que:


$$\Vert{}D_N\Vert{}_1 \sim \log N$$

\medskip
\textbf{Demostración de (6):}
Por la desigualdad triangular:


$$\vert{}D_N(t)\vert{} \le \sum_{k=-N}^N \vert{}e^{ikt}\vert{} = 2N + 1 \implies \Vert{}D_N\Vert{}_\infty \le 2N + 1$$


Como $D_N(0) = 2N + 1$, el supremo se alcanza:


$$\Vert{}D_N\Vert{}_\infty = 2N + 1$$


\end{proof}

\begin{teorema}{}{thm19}
$\exists f \in C(\mathbb{T})$ t.q. $(S_n(f,0))_n$ no converge.
\end{teorema}

\begin{proof}
Utilizamos el Principio de Acotación Uniforme (Banach-Steinhaus). Consideremos:


$$T_n : C(\mathbb{T}) \longrightarrow \mathbb{C}$$


donde se mapea de un espacio vectorial topológico a otro. Si $f \in C(\mathbb{T}) \implies T_n(f) \in \mathbb{C}$ y $(T_n(f))_n \subseteq \mathbb{C}$.

Supongamos que $(S_n(f,0))_n$ converge $\forall f \in C(\mathbb{T})$. Para cada $n \in \mathbb{N}$ considero $T_n: C(\mathbb{T}) \longrightarrow \mathbb{C}$ dada por $T_n(f) = S_n(f,0) = \sum_{k=-n}^n \hat{f}(k)$.

\begin{enumerate}
\item[(a)] $T_n$ es lineal $\forall n$ (inmediato porque $\hat{f}$ es lineal).
\item[(b)] $T_n$ es continua. Si $f_j \xrightarrow{\Vert{}\cdot\Vert{}_\infty} f \implies T_n(f_j) \rightarrow T_n(f)$.
En efecto, si $f_j \xrightarrow{\Vert{}\cdot\Vert{}_\infty} f \implies \hat{f}_j(k) \rightarrow \hat{f}(k) \quad \forall k \implies \sum_{k=-n}^n \hat{f}_j(k) \rightarrow \sum_{k=-n}^n \hat{f}(k) = T_n(f)$.
\item[(c)] Para $f \in C(\mathbb{T})$, $\exists C_f > 0$ t.q $\vert{}T_n(f)\vert{} < C_f \quad \forall n$. Suponemos que $(S_n(f,0))_n$ es convergente, lo que implica que está acotada, por tanto $\vert{}S_n(f,0)\vert{} < C_f \quad \forall n$.
\end{enumerate}

Por el Teorema de Banach-Steinhaus, $\exists C > 0$ t.q. $\Vert{}T_n\Vert{} \le C \quad \forall n$, $\forall f \in C(\mathbb{T})$ con $\Vert{}f\Vert{}_\infty \le 1$. Vamos a ver que:


$$\forall N \in \mathbb{N}, \exists n \in \mathbb{N}, \exists f \in C(\mathbb{T}), \Vert{}f\Vert{}_\infty \le 1 \text{ t.q. } \vert{}S_n(f,0)\vert{} \ge N $$

Fijo $k \in \mathbb{N}$ y $g: [-\pi, \pi] \rightarrow \mathbb{C}$ definida por:


$$g(t) = \begin{cases} 1 & D_k(t) \ge 0 \\ -1 & D_k(t) < 0 \end{cases} $$


Esta función es continua salvo en una cantidad finita de puntos. Puedes construir una sucesión $(g_j)_{j=1}^\infty$ tal que:
\begin{enumerate}
\item[(a)] $g_j: [-\pi, \pi] \rightarrow \mathbb{C}$
\item[(b)] $g_j \in C(\mathbb{T})$
\item[(c)] $-1 \le g_j(t) \le 1 \quad \forall t, \forall j$
\item[(d)] $\lim_{j\to\infty} g_j(t) = g(t) \quad \forall t \in [-\pi, \pi]$
\end{enumerate}

Para cada $j \in \mathbb{N}$ defino $h_j(t) = g_j(t) \cdot D_k(t)$, la cual es continua.
Se cumple que $h(t) = \lim_{j\to\infty} h_j(t) = \lim_{j\to\infty} g_j(t) D_k(t) = \vert{}D_k(t)\vert{}$.
Además, $\vert{}h_j(t)\vert{} \le \vert{}D_k(t)\vert{}$ y $D_k(t)$ es integrable.
Por el Teorema de Convergencia Dominada (TCD) de Lebesgue, $\int_{-\pi}^\pi h(t) dt = \lim_{j\to\infty} \int_{-\pi}^\pi h_j(t) dt$.

Ahora tenemos:

\begin{align*}
\vert{}T_k(g_j)\vert{} = & \vert{}S_k(g_j, 0)\vert{} = \vert{}(g_j * D_k)(0)\vert{} = \left\vert{} \frac{1}{2\pi} \int_{-\pi}^\pi g_j(s) D_k(-s) ds \right\vert{}  =  \frac{1}{2\pi} \left\vert{} \int_{-\pi}^\pi g_j(s) D_k(s) ds \right\vert{} \\ = & \frac{1}{2\pi} \left\vert{} \int_{-\pi}^\pi h_j(s) ds \right\vert{}
\end{align*}
Fijo $k \in \mathbb{N}$:


$$\lim_{j\to\infty} \vert{}S_k(g_j, 0)\vert{} = \frac{1}{2\pi} \lim_{j\to\infty} \left\vert{} \int_{-\pi}^\pi h_j(t) dt \right\vert{} = \frac{1}{2\pi} \int_{-\pi}^\pi \vert{}D_k(s)\vert{} ds = \Vert{}D_k\Vert{}_1 \ge A \log(k)$$

Entonces $\exists j_0$ t.q. $\vert{}S_k(g_j, 0)\vert{} \ge \frac{A \log k}{2} \quad \forall j \ge j_0$.
Dado $N \in \mathbb{N}$:
\begin{itemize}
\item Elijo $n_0$ t.q. $\frac{A \log n_0}{2} > N$
\item Elijo $j_0$ t.q. $\vert{}S_{n_0}(g_{j_0}, 0)\vert{} \ge \frac{A \log n_0}{2}$
\item Sea $f_N = g_{j_0} \in C(\mathbb{T})$, $\Vert{}f_N\Vert{}_\infty \le 1$
\end{itemize}
Con todo, $\vert{}S_{n_0}(f_N, 0)\vert{} \ge N$.

\end{proof}

\begin{observacion}{}{}
El Teorema de Banach-Steinhaus funciona si se reemplaza $C(\mathbb{T})$ por $L_1(\mathbb{T})$ con su norma. Luego, puedo tomar:


$$f_k(t) = \begin{cases} 1 & D_k(t) \ge 0 \\ -1 & D_k(t) < 0 \end{cases} \quad \text{con } f_k \in L_1 $$


\end{observacion}

\textbf{Pregunta:} ¿$(S_N(f))_N$ converge en algún sentido? \textbf{Negativo.}
¿Hay resultados positivos?

\begin{teorema}{M. Weierstraß}{}
Si $f_n: [a,b] \to \mathbb{C}$ son continuas y satisfacen:
\begin{enumerate}
\item $\sup_{t \in [a,b]} \vert{}f_n(t)\vert{} \le M_n$
\item $\sum_{n=1}^\infty M_n < \infty$
\end{enumerate}
Entonces, la serie $\sum_{n=1}^\infty f_n$ converge uniformemente en $[a,b]$.
\end{teorema}

\begin{ejemplo}{Convergencia uniforme}{}
Sea $f(t) = t^2$ en $[-1, 1]$ (extendida periódicamente). Su desarrollo en serie de Fourier es:


$$f(t) \sim \frac{1}{3} + \sum_{\substack{n \in \mathbb{Z} \\ n \neq 0}} \frac{2(-1)^n}{n^2\pi^2} e^{in\pi t}$$


Para los términos $f_n: [-1, 1] \to \mathbb{C}$, que son funciones continuas, tenemos:


$$\sup_{t \in [-1,1]} \vert{}f_n(t)\vert{} = \sup_{t \in [-1,1]} \left\vert{} \frac{2 (-1)^n}{n^2 \pi^2} e^{in\pi t} \right\vert{} = \frac{2}{n^2\pi^2}$$


Tomando $M_n = \frac{2}{n^2\pi^2}$, comprobamos la convergencia de la serie de cotas:
\begin{align*}
\sum_{\substack{n \in \mathbb{Z} \ n \neq 0}} M_n &= \sum_{n=1}^\infty M_n + \sum_{n=-\infty}^{-1} M_n = \sum_{n=1}^\infty M_n + \sum_{n=1}^\infty M_{-n} \
&= \sum_{n=1}^\infty \frac{2}{n^2\pi^2} + \sum_{n=1}^\infty \frac{2}{n^2\pi^2} = \frac{2}{\pi^2} \underbrace{\sum_{n=1}^\infty \frac{1}{n^2}}*{\pi^2/6} + \frac{2}{\pi^2} \underbrace{\sum*{n=1}^\infty \frac{1}{n^2}}_{\pi^2/6} < \infty
\end{align*}
Por tanto, la sucesión de sumas parciales $(S_N(f))_N$ converge uniformemente.
\end{ejemplo}

\begin{observacion}{}{}
Si $\sum_{n \in \mathbb{Z}} \vert{}\hat{f}(n)\vert{}^2 < \infty \Rightarrow$ la serie $(S_n(f))$ converge uniformemente y $f \in \mathcal{C}(\mathbb{T})$.
\end{observacion}

\begin{proposicion}{Lema 1.11 (Lema de Riemann-Lebesgue)}{}
Sea $I \subseteq \mathbb{R}$ un intervalo con $\mathring{I} \neq \emptyset$ y $f \in L_1(I)$.


$$\lim_{\alpha \to \infty} \int_I f(t) \cos(\alpha t + \beta) dt = 0 \quad \forall \beta \in \mathbb{R}$$


\end{proposicion}

\begin{proof}
\textbf{Caso 1:} $\sup [a,b] \subseteq I$ y $f = \chi_{[a,b]}$.
\begin{align*}
\int_I f(t) \cos(\alpha t + \beta) dt &= \int_I \chi_{[a,b]}(t) \cos(\alpha t + \beta) dt = \int_a^b \cos(\alpha t + \beta) dt \
&= \left[ \frac{\sen(\alpha t + \beta)}{\alpha} \right]_a^b = \frac{\sen(\alpha b + \beta)}{\alpha} - \frac{\sen(\alpha a + \beta)}{\alpha}
\end{align*}
Acotando en valor absoluto:


$$0 \le \left\vert{} \int_I \chi_{[a,b]}(t) \cos(\alpha t + \beta) dt \right\vert{} = \frac{\vert{}\sen(\alpha b + \beta) - \sen(\alpha a + \beta)\vert{}}{\alpha} \le \frac{2}{\alpha} \xrightarrow{\alpha \to \infty} 0$$

\textbf{Caso 2:} $f$ es escalonada.
Si $f = \sum_{k=1}^N \lambda_k \chi_{[a_k, b_k]}$, el resultado es cierto por linealidad de la integral y el límite.

\textbf{Caso 3:} $f \in L_1(\mathbb{T})$ arbitraria.
Fijamos $\varepsilon > 0$. Existe una función escalonada $\phi$ tal que $\Vert{}f - \phi\Vert{}_1 < \varepsilon/2$, es decir:


$$\int_I \vert{}f(t) - \phi(t)\vert{} dt < \frac{\varepsilon}{2}$$


Por el Caso 2, existe $\alpha_0 > 0$ tal que para $\alpha > \alpha_0$:


$$\left\vert{} \int_I \phi(t) \cos(\alpha t + \beta) dt \right\vert{} < \frac{\varepsilon}{2}$$


Por tanto, utilizando la desigualdad triangular:
\begin{align*}
\left| \int_I f(t) \cos(\alpha t + \beta) dt \right| &= \left| \int_I (f(t) - \phi(t) + \phi(t)) \cos(\alpha t + \beta) dt \right| \
&\le \left| \int_I (f(t) - \phi(t)) \cos(\alpha t + \beta) dt \right| + \underbrace{\left| \int_I \phi(t) \cos(\alpha t + \beta) dt \right|}_{< \varepsilon/2} \
&\le \int_I |f(t) - \phi(t)| |\cos(\alpha t + \beta)| dt + \frac{\varepsilon}{2} \
&\le \int_I |f(t) - \phi(t)| dt + \frac{\varepsilon}{2} < \frac{\varepsilon}{2} + \frac{\varepsilon}{2} = \varepsilon
\end{align*}
\end{proof}

\begin{teorema}{Teorema 1.12}{}
Sea $f \in L_1(\mathbb{T})$ y $0 < \delta < \pi$. Entonces $S_n(f,t)$ converge si y solo si:


$$\exists \lim_{n \to \infty} \frac{1}{2\pi} \int_0^\delta (f(t+s) + f(t-s)) D_n(s) ds$$


Además, en este caso, la serie $S_n(f,t)$ converge a dicho límite.
\end{teorema}

\begin{proof}
Usando la convolución con el núcleo de Dirichlet $D_n(s)$:
\begin{align*}
S_n(f,t) &= (D_n * f)(t) = \frac{1}{2\pi} \int_{-\pi}^\pi f(t-s) D_n(s) ds \
&= \frac{1}{2\pi} \left( \int_{-\pi}^0 f(t-s) D_n(s) ds + \int_0^\pi f(t-s) D_n(s) ds \right)
\end{align*}
Haciendo el cambio de variable $u = -s$ en la primera integral y usando que $D_n$ es par ($D_n(-u) = D_n(u)$):
\begin{align*}
S_n(f,t) &= \frac{1}{2\pi} \left( \int_0^\pi f(t+s) D_n(s) ds + \int_0^\pi f(t-s) D_n(s) ds \right) \
&= \frac{1}{2\pi} \int_0^\pi (f(t+s) + f(t-s)) D_n(s) ds \
&= \frac{1}{2\pi} \left( \int_0^\delta (f(t+s) + f(t-s)) D_n(s) ds + \int_\delta^\pi (f(t+s) + f(t-s)) D_n(s) ds \right)
\end{align*}
Veamos que el segundo término tiende a cero cuando $n \to \infty$:


$$\lim_{n \to \infty} \int_\delta^\pi (f(t+s) + f(t-s)) D_n(s) ds = 0 \quad (*)$$


Sustituyendo el núcleo de Dirichlet $D_n(s) = \frac{\sen((n+1/2)s)}{\sen(s/2)}$:


$$\int_\delta^\pi (f(t+s) + f(t-s)) \frac{\sen((n+1/2)s)}{\sen(s/2)} ds = \int_\delta^\pi \varphi(s) \sen((n+1/2)s) ds$$


Donde $\varphi(s) = \frac{f(t+s) + f(t-s)}{\sen(s/2)}$. Al ser $\varphi \in L_1([\delta, \pi])$ (como se justifica en la observación posterior), esta integral tiende a 0 por el Lema de Riemann-Lebesgue.

Sabiendo que (*) es cierto, $S_n(f,t)$ convergerá si y solo si la integral sobre $[0, \delta]$ converge.

Estudiamos el comportamiento del límite aislando los valores laterales. Si sumamos y restamos $f(t+)$ y $f(t-)$, el término queda descompuesto en tres integrales. Nos enfocamos en la tercera:
\begin{align*}
\int_0^\delta (f(t+) + f(t-)) D_n(s) ds &= (f(t+) + f(t-)) \left( \int_0^\pi D_n(s) ds - \int_\delta^\pi D_n(s) ds \right) \
&= (f(t+) + f(t-)) \left( \frac{1}{2} - \frac{1}{\pi} \int_\delta^\pi \frac{\sen((n+1/2)s)}{\sen(s/2)} ds \right)
\end{align*}
Aplicando Riemann-Lebesgue a la última integral obtenemos que tiende a 0. Así:


$$\lim_{n \to \infty} \frac{1}{2\pi} \int_0^\delta (f(t+s) + f(t-s)) D_n(s) ds = \frac{f(t+) + f(t-)}{2}$$


Las dos integrales restantes tienden a 0 asumiendo condiciones de integrabilidad local (ver Test de Dini).
\end{proof}

\begin{observacion}{}{}
\begin{itemize}
\item La función $\sen(x)$ es creciente y $\sen(x) \ge 0$ para todo $x \in [0, \pi/2]$.
\item Para $\varphi(s) = \frac{f(t+s) + f(t-s)}{\sen(s/2)}$, si $0 < \delta \le s \le \pi \implies 0 < \delta/2 \le s/2 \le \pi/2 \implies \sen(s/2) \neq 0$ en $[\delta, \pi]$.
\end{itemize}
Además, $\vert{}\varphi(s)\vert{} \le \frac{\vert{}f(t+s) + f(t-s)\vert{}}{\sen(\delta/2)}$, que es integrable. Luego $\varphi \in L_1([\delta, \pi])$.
\end{observacion}

\begin{teorema}{Test de Dini}{}
Sean $f \in L_1(\mathbb{T})$ y $t \in \mathbb{R}$ tales que existen los límites laterales:


$$f(t-) = \lim_{s \to t^-} f(s) \quad \text{y} \quad f(t+) = \lim_{s \to t^+} f(s)$$


Si existe un $\delta > 0$ tal que las siguientes integrales son convergentes:


$$\int_0^\delta \frac{f(t+s) - f(t+)}{s} ds \quad \text{y} \quad \int_0^\delta \frac{f(t-s) - f(t-)}{s} ds$$


Entonces:


$$\exists \lim_{n \to \infty} S_n(f,t) = \frac{f(t+) + f(t-)}{2}$$


\end{teorema}

\begin{proof}
Por el Teorema 1.12, sabemos que $\lim_{n \to \infty} S_n(f,t) = L \iff \lim_{n \to \infty} \frac{1}{2\pi} \int_0^\delta (f(t+s) + f(t-s)) D_n(s) ds = L$.
Descomponiendo la integral:
\begin{align*}
\int_0^\delta (f(t+s) + f(t-s)) D_n(s) ds &= \int_0^\delta (f(t+s) - f(t+)) D_n(s) ds \
&\quad + \int_0^\delta (f(t-s) - f(t-)) D_n(s) ds \
&\quad + \int_0^\delta (f(t+) + f(t-)) D_n(s) ds
\end{align*}
Reescribiendo la primera de estas tres integrales:


$$\int_0^\delta (f(t+s) - f(t+)) D_n(s) ds = \int_0^\delta \underbrace{\left[ \frac{f(t+s) - f(t+)}{s} \right]}_{\mathcal{C}_1} \underbrace{\frac{s}{\sen(s/2)}}_{\mathcal{C}_2} \sen((n+1/2)s) ds$$


Dado que $\lim_{s \to 0} \frac{s}{\sen(s/2)} = 2$, la función $\mathcal{C}_2$ es acotada e integrable en $[0, \delta]$. Como por hipótesis $\mathcal{C}_1$ es integrable, el producto $\mathcal{C}_1 \cdot \mathcal{C}_2$ es integrable en $[0, \delta]$.
Aplicando el Lema de Riemann-Lebesgue, esta integral tiende a 0. Análogamente, la segunda integral tiende a 0.
Como ya se demostró en el Teorema 1.12, la tercera integral tiende al valor $\frac{f(t+) + f(t-)}{2}$.
\end{proof}

\begin{proposicion}{Corolario 1.14}{}
Sea $f \in L_1(\mathbb{T})$ y $t \in \mathbb{R}$ tal que existen $f(t+)$ y $f(t-)$. Supongamos que existen las derivadas laterales:


$$f'_-(t) = \lim_{s \to 0^-} \frac{f(t+s) - f(t-)}{s} \quad \text{y} \quad f'_+(t) = \lim_{s \to 0^+} \frac{f(t+s) - f(t+)}{s}$$


Entonces:


$$S_n(f,t) \xrightarrow{n \to \infty} \frac{f(t+) + f(t-)}{2}$$


\end{proposicion}

\begin{proof}
Puesto que existe el límite $f'_-(t)$, fijado un valor límite, existe un $\delta > 0$ tal que para todo $0 < s \le \delta$:


$$\left\vert{} \frac{f(t-s) - f(t-)}{-s} - f'_-(t) \right\vert{} < 1$$


De aquí se deduce que:


$$\left\vert{} \frac{f(t-s) - f(t-)}{s} \right\vert{} \le 1 + \vert{}f'_-(t)\vert{}$$


La función integrando es acotada en $[0, \delta]$, y por tanto, integrable. Razonando de forma análoga con $f'_+(t)$, se satisfacen plenamente las condiciones exigidas por el Test de Dini, garantizando la convergencia del desarrollo de Fourier a la semisuma de los límites laterales.
\end{proof}

\subsubsection{Aplicaciones del test de Dini}

\begin{nota}{Ubicación de estos ejemplos según el temario}{ubic_ejemplos_dini}
Los tres ejemplos siguientes aplican el Corolario 1.14 (y el test de Dini) y sirven para calcular sumas de series numéricas. No aparecen en el pdf de teoría (AF\_1), donde tras el Corolario 1.14 viene directamente la proposición sobre funciones con condición de Hölder y después Cesàro. Por eso se dejan aquí, justo detrás del Corolario 1.14, como ejemplos de clase.
\end{nota}

\begin{ejemplo}{Función signo: Dini en $t=0$ y $t=\pi/2$}{dini_signo}
Sea
\[
f(t) = \begin{cases} 1, & 0 \le t < \pi, \\ -1, & -\pi \le t < 0, \end{cases}
\]
extendida $2\pi$-periódicamente. Su serie de Fourier es
\[
f(t) \sim \frac{4}{\pi} \sum_{k=1}^\infty \frac{1}{2k-1} \operatorname{sen}\big((2k-1)t\big).
\]

\textbf{Caso $t=0$.} Existen los límites laterales
\[
f(0+) = \lim_{t \to 0^+} f(t) = 1, \qquad f(0-) = \lim_{t \to 0^-} f(t) = -1,
\]
luego se cumplen las dos primeras condiciones del test de Dini. Además existen las derivadas laterales:
\[
f'_+(0) = \lim_{s \to 0^+} \frac{f(0+s) - f(0+)}{s} = \lim_{s \to 0^+} \frac{1-1}{s} = 0, \qquad f'_-(0) = \lim_{s \to 0^-} \frac{f(0+s) - f(0-)}{s} = \lim_{s \to 0^-} \frac{-1-(-1)}{s} = 0.
\]
Podemos aplicar el Corolario 1.14:
\[
S_n(f,0) \longrightarrow \frac{f(0+) + f(0-)}{2} = \frac{1 + (-1)}{2} = 0.
\]
De hecho, comprobándolo con la serie, $\operatorname{sen}((2k-1)\cdot 0) = 0$, luego $S_n(f,0) = 0$ para todo $n$.

\textbf{Caso $t=\pi/2$.} $f$ es continua y derivable en $\pi/2$ (es constante igual a $1$ en un entorno), así que existen los límites laterales
\[
\lim_{t \to \frac{\pi}{2}^-} f(t) = 1 = f\!\left(\tfrac{\pi}{2}-\right), \qquad \lim_{t \to \frac{\pi}{2}^+} f(t) = 1 = f\!\left(\tfrac{\pi}{2}+\right),
\]
y las derivadas laterales valen $0$. Por Dini,
\[
S_n\!\left(f,\tfrac{\pi}{2}\right) \to \frac{f\left(\frac{\pi}{2}+\right) + f\left(\frac{\pi}{2}-\right)}{2} = \frac{1+1}{2} = 1 = f\!\left(\tfrac{\pi}{2}\right).
\]
Comprobémoslo con la serie: como $\operatorname{sen}\!\big((2k-1)\tfrac{\pi}{2}\big) = \operatorname{sen}\!\big(k\pi - \tfrac{\pi}{2}\big) = -\cos(k\pi) = (-1)^{k+1}$,
\[
1 = f\!\left(\tfrac{\pi}{2}\right) = S\!\left(f,\tfrac{\pi}{2}\right) = \frac{4}{\pi} \sum_{k=1}^\infty \frac{(-1)^{k+1}}{2k-1} \implies \sum_{k=1}^\infty \frac{(-1)^{k+1}}{2k-1} = \frac{\pi}{4}.
\]
\end{ejemplo}

\begin{ejemplo}{$f(t)=t^2$ en $[-1,1[$: Dini en $t=0$}{dini_t2_simetrico}
Sea $f(t) = t^2$ para $-1 \le t < 1$, extendida con periodo $2$. Su serie (ver el ejemplo del 24/09) es
\[
f(t) \sim \frac{1}{3} + \sum_{n=1}^\infty \frac{4(-1)^n}{n^2\pi^2} \cos(n\pi t).
\]
En $t=0$ la función es continua y derivable, luego por Dini $S_n(f,0) \to f(0) = 0$. Evaluando la serie en $t=0$:
\[
0 = f(0) = \frac{1}{3} + \sum_{n=1}^\infty \frac{4(-1)^n}{n^2\pi^2} \implies -\frac{1}{3} = \frac{4}{\pi^2}\sum_{n=1}^\infty \frac{(-1)^n}{n^2} \implies \sum_{n=1}^\infty \frac{(-1)^{n+1}}{n^2} = \frac{\pi^2}{12}.
\]
\end{ejemplo}

\begin{ejemplo}{$f(t)=t^2$ en $[0,2\pi[$: Dini en $t=0$}{dini_t2_2pi}
Sea ahora $f(t) = t^2$ para $0 \le t < 2\pi$, extendida con periodo $2\pi$.

\textbf{Coeficientes.} Como $f$ es $2\pi$-periódica, se puede integrar en $[0,2\pi]$ en lugar de $[-\pi,\pi]$. (En clase surgió la pregunta de por qué cambian los límites: es válido precisamente por la periodicidad, ver la nota de integración en un periodo.)
\[
\hat{f}(0) = \frac{1}{2\pi} \int_0^{2\pi} t^2 \, dt = \frac{1}{2\pi} \cdot \frac{(2\pi)^3}{3} = \frac{4\pi^2}{3}.
\]
Para $n \ne 0$, integrando por partes dos veces,
\[
\int t^2 e^{-int} dt = -e^{-int}\left( \frac{t^2}{in} + \frac{2t}{(in)^2} + \frac{2}{(in)^3} \right),
\]
y evaluando entre $0$ y $2\pi$ (usando $e^{-2\pi i n} = 1$) resulta
\[
\hat{f}(n) = \frac{2}{n^2} + \frac{2\pi i}{n}.
\]
Pasando a coeficientes reales,
\[
a_n = \hat{f}(n) + \hat{f}(-n) = \frac{4}{n^2}, \qquad b_n = i\big(\hat{f}(n) - \hat{f}(-n)\big) = i \cdot \frac{4\pi i}{n} = -\frac{4\pi}{n},
\]
y $a_0/2 = \hat{f}(0) = 4\pi^2/3$.

\textbf{Aplicación de Dini en $t=0$.} Los límites laterales en $0$ son
\[
f(0+) = \lim_{t \to 0^+} t^2 = 0, \qquad f(0-) = \lim_{t \to 0^-} f(t) = \lim_{t \to 2\pi^-} t^2 = 4\pi^2 \quad \text{(por ser $2\pi$-periódica)},
\]
y existen las derivadas laterales:
$f'_+(0) = \lim_{s \to 0^+} \frac{f(0+s)-0}{s} = \lim_{s\to 0^+} s = 0$ y
$f'_-(0) = \lim_{s \to 0^-} \frac{f(0+s) - 4\pi^2}{s} = \lim_{s\to 0^-} \frac{(2\pi+s)^2 - 4\pi^2}{s} = 4\pi$
(existe porque $x \mapsto x^2$ es derivable en $2\pi$). Por el Corolario 1.14,
\[
S_n(f,0) \to \frac{f(0+) + f(0-)}{2} = \frac{0 + 4\pi^2}{2} = 2\pi^2.
\]
Evaluando la serie en $t=0$ ($\cos 0 = 1$, $\operatorname{sen} 0 = 0$):
\[
2\pi^2 = \frac{4\pi^2}{3} + \sum_{n=1}^\infty \frac{4}{n^2} \implies \sum_{n=1}^\infty \frac{4}{n^2} = 2\pi^2 - \frac{4\pi^2}{3} = \frac{2\pi^2}{3} \implies \sum_{n=1}^\infty \frac{1}{n^2} = \frac{\pi^2}{6}.
\]
\end{ejemplo}

\subsection{Convergencia de Cesàro}

\subsubsection{Sucesiones y series escalares}

\begin{definicion}{Sucesión Cesàro convergente}{cesaro_sucesion}
Una sucesión $(\lambda_n)_n \subseteq \mathbb{C}$ es \textbf{Cesàro convergente} a $\lambda \in \mathbb{C}$ si la sucesión de medias aritméticas converge a $\lambda$:
\[
\frac{1}{n}(\lambda_1 + \lambda_2 + \cdots + \lambda_n) \xrightarrow{n\to\infty} \lambda.
\]
\end{definicion}

\begin{proposicion}{Proposición 1.17}{prop_1_17}
Si $(\lambda_n)_n \to \lambda$, entonces $(\lambda_n)_n$ converge Cesàro a $\lambda$.
\end{proposicion}

\begin{proof}
Hay que estudiar $\left| \frac{1}{n} \sum_{k=1}^n \lambda_k - \lambda \right|$ y ver que tiende a cero. Se tiene
\[
\left| \frac{1}{n} \sum_{k=1}^n \lambda_k - \lambda \right| = \left| \sum_{k=1}^n \frac{\lambda_k - \lambda}{n} \right|.
\]
Fijo $\varepsilon > 0$. Existe $n_0 \in \mathbb{N}$ tal que $|\lambda_n - \lambda| < \varepsilon/2$ para todo $n \ge n_0$. Como $(\lambda_n)$ converge, está acotada: existe $M > 0$ con $|\lambda_n| < M$ para todo $n$, y por tanto $|\lambda_k - \lambda| \le |\lambda_k| + |\lambda| \le 2M$ (tomando $M$ también mayor que $|\lambda|$).

Entonces, si $n \ge n_0$, separando la suma en dos partes:
\[
\left| \sum_{k=1}^n \frac{\lambda_k - \lambda}{n} \right| \le \sum_{k=1}^{n_0} \frac{|\lambda_k - \lambda|}{n} + \sum_{k=n_0+1}^n \frac{|\lambda_k - \lambda|}{n} \le \sum_{k=1}^{n_0} \frac{2M}{n} + \sum_{k=n_0+1}^n \frac{\varepsilon/2}{n} \le \frac{2M n_0}{n} + \frac{\varepsilon}{2}.
\]
Como $\frac{2Mn_0}{n} \to 0$, existe $n_1$ tal que $\frac{2Mn_0}{n} < \varepsilon/2$ para todo $n \ge n_1$. Entonces, si $n \ge \max(n_0,n_1)$,
\[
\left| \frac{1}{n} \sum_{k=1}^n \lambda_k - \lambda \right| < \frac{\varepsilon}{2} + \frac{\varepsilon}{2} = \varepsilon,
\]
es decir, $\lim_{n\to\infty} \frac{1}{n}\sum_{k=1}^n \lambda_k = \lambda$.
\end{proof}

\begin{ejemplo}{La sucesión $(-1)^n$}{ej_cesaro_menos_uno}
La sucesión $(\lambda_n) = ((-1)^n)$ \textbf{no converge}. Veamos que sí converge Cesàro. Sus sumas parciales (sumando desde $n=1$) son
\[
\lambda_1 = -1, \quad \lambda_1 + \lambda_2 = 0, \quad \lambda_1+\lambda_2+\lambda_3 = -1, \quad \lambda_1+\lambda_2+\lambda_3+\lambda_4 = 0, \dots
\]
es decir
\[
\sum_{k=1}^n \lambda_k = \begin{cases} 0, & n \text{ par}, \\ -1, & n \text{ impar}, \end{cases} \quad \Longrightarrow \quad 0 \le \left| \frac{1}{n} \sum_{k=1}^n \lambda_k \right| \le \frac{1}{n} \to 0.
\]
Luego $((-1)^n)_n$ es convergente Cesàro (y converge a $0$). Esto muestra que el recíproco de la Proposición 1.17 es falso.
\end{ejemplo}

\begin{definicion}{Serie Cesàro convergente}{cesaro_serie}
Dada $(\lambda_n)_{n=1}^\infty$, se considera la sucesión de sumas parciales $s_1 = \lambda_1$, $s_2 = \lambda_1 + \lambda_2$, $s_3 = \lambda_1+\lambda_2+\lambda_3, \dots$, es decir $s_N = \sum_{k=1}^N \lambda_k$. Si $(s_N)_N$ converge Cesàro, decimos que la serie $\sum_k \lambda_k$ \textbf{converge Cesàro}. Es decir, si
\[
\sigma_N = \frac{1}{N}\left( s_1 + s_2 + \cdots + s_N \right) \quad \text{converge.}
\]
\end{definicion}

\begin{nota}{Índices desde 0 y fórmula de $\sigma_N$}{nota_indices_cesaro}
Cuando se empieza con $(\lambda_n)_{n=0}^\infty$ las sumas parciales son $s_0 = \lambda_0$, $s_1 = \lambda_0+\lambda_1, \dots, s_N = \lambda_0 + \cdots + \lambda_N$, y
\[
\sigma_0 = s_0, \quad \sigma_1 = \tfrac{1}{2}(s_0 + s_1), \quad \sigma_2 = \tfrac{1}{3}(s_0+s_1+s_2), \quad \dots, \quad \sigma_N = \frac{1}{N+1}(s_0 + s_1 + \cdots + s_N).
\]
Si esto converge, $\sum \lambda_n$ es Cesàro convergente. Desarrollando los primeros términos:
\[
\sigma_1 = \tfrac{1}{2}(2\lambda_0 + \lambda_1), \quad \sigma_2 = \tfrac{1}{3}(3\lambda_0 + 2\lambda_1 + \lambda_2), \quad \sigma_3 = \tfrac{1}{4}(4\lambda_0 + 3\lambda_1 + 2\lambda_2 + \lambda_3),
\]
lo que sugiere
\[
\sigma_N = \frac{1}{N+1} \sum_{k=0}^N (N+1-k)\lambda_k = \sum_{k=0}^N \left(1 - \frac{k}{N+1}\right) \lambda_k.
\]
Se demostraría por inducción (\textbf{ejercicio importante}) que esta fórmula es cierta. Es la que se usará para las sumas de Fourier.
\end{nota}

\begin{ejemplo}{Motivación: pulsos $f_\delta$ y sus coeficientes}{ej_pulso_f_delta}
Dado $0 < \delta < \pi$, sea $f_\delta : [-\pi,\pi] \to \mathbb{C}$,
\[
f_\delta(t) = \begin{cases} \dfrac{1}{2\delta}, & -\delta \le t \le \delta, \\[1ex] 0, & \text{otro caso.} \end{cases}
\]
Sus coeficientes de Fourier son
\[
\hat{f}_\delta(0) = \frac{1}{2\pi} \int_{-\delta}^{\delta} \frac{1}{2\delta} \, dt = \frac{1}{2\pi},
\]
y para $n \ne 0$,
\[
\hat{f}_\delta(n) = \frac{1}{2\pi} \int_{-\delta}^{\delta} \frac{1}{2\delta} e^{-int} dt = \frac{1}{2\pi} \cdot \frac{1}{2\delta} \cdot \frac{-1}{in} \left( e^{-in\delta} - e^{in\delta} \right) = \frac{1}{2\pi} \cdot \frac{\operatorname{sen}(n\delta)}{n\delta}.
\]
Cuando $\delta \to 0$, $f_\delta$ tiende a ``$\infty$'' en $t=0$ y a $0$ en $t \ne 0$, mientras que
\[
\lim_{\delta \to 0} \hat{f}_\delta(n) = \frac{1}{2\pi} \quad \forall n \in \mathbb{Z}.
\]
\end{ejemplo}

\begin{ejemplo}{Cesàro para sucesiones: $1,-1,1,\dots$}{ej_cesaro_media}
Sean $\lambda_0, \lambda_1, \ldots \in \mathbb{C}$ una sucesión. ¿Converge?

\begin{enumerate}
\item[(a)] Convergencia ``normal'': $\forall \varepsilon > 0$, $\exists n_0 \in \mathbb{N}$ tal que \dots
\item[(b)] Cesàro: tomo las medias aritméticas
\[
m_0 = \lambda_0, \quad m_1 = \frac{\lambda_0 + \lambda_1}{2}, \quad m_2 = \frac{1}{3}(\lambda_0 + \lambda_1 + \lambda_2), \quad \dots, \quad m_n = \frac{1}{n+1}(\lambda_0 + \cdots + \lambda_n),
\]
y la pregunta es si $(m_n)_n$ converge.
\end{enumerate}

Para $\lambda_n = (-1)^n$ ($1,-1,1,-1,\dots$): no converge, pero $m_0 = 1$, $m_1 = 0$, $m_2 = \tfrac{1}{3}$, \dots,
\[
m_n = \begin{cases} \dfrac{1}{n+1}, & n \text{ par}, \\[1ex] 0, & n \text{ impar}, \end{cases} \quad \longrightarrow 0,
\]
luego $(-1)^n \xrightarrow{\,n\to\infty\,} 0$ en el sentido de Cesàro.
\end{ejemplo}

\begin{ejemplo}{La serie $\sum (-1)^n$ en el sentido de Cesàro}{ej_cesaro_serie}
Dada una serie $\lambda_0 + \lambda_1 + \cdots + \lambda_n + \cdots$, ¿cuánto suma? Se considera la sucesión de sumas parciales $s_0 = \lambda_0$, $s_1 = \lambda_0 + \lambda_1$, $s_2 = \lambda_0+\lambda_1+\lambda_2$, \dots, $s_n = \sum_{k=0}^n \lambda_k$.
\begin{enumerate}
\item[(a)] $s_n \to L \iff \sum \lambda_n$ converge y $\sum_{n=0}^\infty \lambda_n = L$.
\item[(b)] Cesàro: $s_n \xrightarrow{\text{Cesàro}} L \iff \mathcal{C}\text{-}\sum \lambda_n = L$. Es decir, las medias aritméticas de $(s_n)$:
\[
\sigma_0 = s_0, \quad \sigma_1 = \tfrac{1}{2}(s_0+s_1), \quad \sigma_2 = \tfrac{1}{3}(s_0+s_1+s_2), \quad \dots, \quad \sigma_n = \tfrac{1}{n+1}(s_0+\cdots+s_n),
\]
verifican $\sigma_n \to L \iff s_n \xrightarrow{\mathcal{C}} L \iff \mathcal{C}\text{-}\sum \lambda_n = L$.
\end{enumerate}

Para $1,-1,1,-1,\dots$ las sumas parciales son $s_0 = 1$, $s_1 = 0$, $s_2 = 1$, \dots, y
\[
\sigma_0 = 1, \quad \sigma_1 = \tfrac{1}{2}, \quad \sigma_2 = \tfrac{2}{3}, \quad \sigma_3 = \tfrac{1}{2}, \quad \sigma_4 = \tfrac{3}{5}, \quad \sigma_5 = \tfrac{1}{2}, \qquad \sigma_n = \begin{cases} \tfrac{1}{2}, & n \text{ impar}, \\[0.5ex] \dfrac{n/2+1}{n+1}, & n \text{ par}, \end{cases} \quad \xrightarrow{n\to\infty} \tfrac{1}{2}.
\]
Luego $\displaystyle\sum_{n=0}^\infty (-1)^n = \tfrac{1}{2}$ en el sentido de Cesàro. Usando la fórmula de la nota anterior, $\sigma_n = \frac{1}{n+1}\sum_{k=0}^n (n+1-k)\lambda_k$.
\end{ejemplo}

\subsection{Medias de Cesàro de una serie de Fourier}

\textbf{Idea.} Si $f \in L_1$ tengo su serie de Fourier, que no tiene por qué converger. Aplico el ``método de Cesàro'' a la sucesión de sumas parciales $(S_n(f,t))_n$.

\begin{definicion}{Media de Cesàro de $f$}{media_cesaro_f}
Si $f \in L_1(\mathbb{T})$ y $N \in \mathbb{N}$, se define la $N$-ésima media de Cesàro de $f$ como
\[
\sigma_N(f,t) = \sigma_N(f)(t) = \frac{1}{N+1}\Big( S_0(f,t) + S_1(f,t) + \cdots + S_N(f,t) \Big).
\]
\end{definicion}

\textbf{Pregunta:} ¿$\sigma_N(f) \to f$ en algún sentido? Observamos primero que, escribiendo $S_n(f,t) = \sum_{k=-n}^n \hat{f}(k) e^{ikt} = \frac{a_0}{2} + \sum_{k=1}^n \underbrace{\big(a_k \cos(kt) + b_k \operatorname{sen}(kt)\big)}_{\lambda_k}$, la media de Cesàro es la media de las sumas parciales de $\sum \lambda_k$, y por la fórmula de la nota anterior:
\[
\sigma_N(f,t) = \frac{a_0}{2} + \sum_{k=1}^N \left(1 - \frac{k}{N+1}\right)\big(a_k \cos(kt) + b_k \operatorname{sen}(kt)\big) = \sum_{k=-N}^{N} \left(1 - \frac{|k|}{N+1}\right) \hat{f}(k) e^{ikt}.
\]

\textbf{Paso 1: escribir $\sigma_N(f)$ como una convolución.} Usando $S_n(f,t) = (f * D_n)(t)$:
\[
\sigma_N(f,t) = \frac{1}{N+1}\sum_{n=0}^N S_n(f,t) = \frac{1}{N+1}\sum_{n=0}^N \frac{1}{2\pi}\int_{-\pi}^{\pi} f(s) D_n(t-s)\, ds = \frac{1}{2\pi}\int_{-\pi}^{\pi} f(s) \underbrace{\frac{1}{N+1}\sum_{n=0}^N D_n(t-s)}_{F_N(t-s)} ds = (f * F_N)(t).
\]

\subsubsection{El núcleo de Fejér}

\begin{definicion}{Núcleo de Fejér}{nucleo_fejer}
Para cada $N \in \mathbb{N}$ se define el \textbf{núcleo de Fejér} $N$-ésimo como
\[
F_N(t) = \frac{1}{N+1}\sum_{n=0}^N D_n(t).
\]
Por lo anterior, $\sigma_N(f,t) = (f * F_N)(t)$. (Observación de clase: se suele escribir también $(K_N)_N$, \emph{Fejér kernel}.)
\end{definicion}

\begin{proposicion}{Proposición 1.18: propiedades del núcleo de Fejér}{prop_1_18}
$F_N$ es $2\pi$-periódica, continua, no negativa ($F_N(t) \ge 0$) y par ($F_N(t) = F_N(-t)$). Además
\[
F_N(t) = \begin{cases} N+1, & t = 2k\pi, \; k \in \mathbb{Z}, \\[1ex] \dfrac{1}{N+1}\left( \dfrac{\operatorname{sen}\left(\frac{N+1}{2}t\right)}{\operatorname{sen}(t/2)} \right)^2, & \text{otro caso.} \end{cases}
\]
\end{proposicion}

\begin{proof}
\begin{itemize}
\item \textbf{$2\pi$-periódica:} obvio, es una suma finita de funciones $2\pi$-periódicas.
\item \textbf{Continua:} suma finita de funciones continuas.
\item \textbf{Par:} suma finita de funciones pares.
\item $F_N(t) \ge 0$: es obvio a partir de la descripción (un cuadrado dividido por $N+1>0$).
\end{itemize}
\textbf{Caso $t = 2k\pi$, $k \in \mathbb{Z}$.} Como $D_n(2k\pi) = 2n+1$,
\[
F_N(2k\pi) = \frac{1}{N+1}\sum_{n=0}^N (2n+1) = \frac{1}{N+1}\left( 2\sum_{n=0}^N n + \sum_{n=0}^N 1 \right) = \frac{1}{N+1}\left( N(N+1) + (N+1) \right) = N+1.
\]
\textbf{Otro $t$.} Usando $D_n(t) = \frac{\operatorname{sen}((n+\frac{1}{2})t)}{\operatorname{sen}(t/2)}$:
\[
F_N(t) = \frac{1}{N+1}\sum_{n=0}^N \frac{\operatorname{sen}((n+\frac{1}{2})t)}{\operatorname{sen}(t/2)} = \frac{1}{N+1} \cdot \frac{1}{\operatorname{sen}(t/2)} \cdot \operatorname{Im}\left( \sum_{n=0}^N e^{i(n+\frac{1}{2})t} \right),
\]
ya que $e^{ia} = \cos a + i\operatorname{sen} a$ y $\operatorname{sen} a = \operatorname{Im}(e^{ia})$. Sumando la progresión geométrica:
\begin{align*}
\sum_{n=0}^N e^{i(n+\frac{1}{2})t} &= e^{it/2}\sum_{n=0}^N (e^{it})^n = e^{it/2}\,\frac{(e^{it})^{N+1} - 1}{e^{it} - 1} = e^{it/2}\,\frac{e^{i\frac{N+1}{2}t}\left( e^{i\frac{N+1}{2}t} - e^{-i\frac{N+1}{2}t} \right)}{e^{it/2}\left( e^{it/2} - e^{-it/2} \right)} \\
&= e^{i\frac{N+1}{2}t} \,\frac{\operatorname{sen}\left(\frac{N+1}{2}t\right)}{\operatorname{sen}(t/2)} = \left( \cos\left(\tfrac{N+1}{2}t\right) + i\operatorname{sen}\left(\tfrac{N+1}{2}t\right) \right)\frac{\operatorname{sen}\left(\frac{N+1}{2}t\right)}{\operatorname{sen}(t/2)}.
\end{align*}
Tomando la parte imaginaria,
\[
\operatorname{Im}\left( \sum_{n=0}^N e^{i(n+\frac{1}{2})t} \right) = \frac{\left(\operatorname{sen}\left(\frac{N+1}{2}t\right)\right)^2}{\operatorname{sen}(t/2)},
\]
y por tanto
\[
F_N(t) = \frac{1}{N+1}\cdot\frac{1}{\operatorname{sen}(t/2)}\cdot\frac{\left(\operatorname{sen}\left(\frac{N+1}{2}t\right)\right)^2}{\operatorname{sen}(t/2)} = \frac{1}{N+1}\left( \frac{\operatorname{sen}\left(\frac{N+1}{2}t\right)}{\operatorname{sen}(t/2)} \right)^2. \qedhere
\]
\end{proof}

\begin{teorema}{Teorema 1.19: núcleo de Fejér}{thm_1_19}
\begin{enumerate}
\item[(1)] $\displaystyle\frac{1}{2\pi}\int_{-\pi}^{\pi} F_N(t)\,dt = 1$ para todo $N \in \mathbb{N}$.
\item[(2)] $\displaystyle F_N(t) \le \frac{1}{N+1}\,\frac{1}{(\operatorname{sen}(\delta/2))^2}$ para todo $\delta < |t| \le \pi$.
\item[(3)] $F_N \xrightarrow{N\to\infty} 0$ uniformemente en $[-\pi,-\delta]\cup[\delta,\pi]$, para todo $\delta$.
\item[(4)] $\displaystyle\lim_{N}\left( \frac{1}{2\pi}\int_{\delta < |t| < \pi} F_N(t)\,dt \right) = 0$, para todo $\delta$.
\end{enumerate}
\end{teorema}

\begin{proof}
\textbf{(1)} Por la propiedad (4) del núcleo de Dirichlet,
\[
\frac{1}{2\pi}\int_{-\pi}^{\pi} F_N(t)\,dt = \frac{1}{2\pi}\int_{-\pi}^{\pi}\frac{1}{N+1}\sum_{n=0}^N D_n(t)\,dt = \frac{1}{N+1}\sum_{n=0}^N \underbrace{\frac{1}{2\pi}\int_{-\pi}^{\pi} D_n(t)\,dt}_{1} = \frac{1}{N+1}(N+1) = 1.
\]
\textbf{(2)} Fijo $\delta > 0$ y tomo $\delta < |t| \le \pi$ (en particular $t \notin \{2k\pi\}$). Entonces
\[
0 \le F_N(t) = \frac{1}{N+1}\left( \frac{\operatorname{sen}\left(\frac{N+1}{2}t\right)}{\operatorname{sen}(t/2)} \right)^2 \le \frac{1}{N+1}\,\frac{1}{(\operatorname{sen}(t/2))^2} \le \frac{1}{N+1}\,\frac{1}{(\operatorname{sen}(\delta/2))^2},
\]
donde la última desigualdad se debe a que $\operatorname{sen}$ es creciente en $[0,\pi/2]$: de $\delta < |t| \le \pi$ se deduce $\delta/2 < |t|/2 \le \pi/2$ y $\operatorname{sen}(|t|/2) > \operatorname{sen}(\delta/2)$.

\textbf{(3)} y \textbf{(4)} se deducen de (2): la cota depende de $N$ solo a través de $\frac{1}{N+1} \to 0$, y no depende de $t$ (de ahí la convergencia uniforme); además la integral en $\{\delta<|t|<\pi\}$ está acotada por $\frac{1}{2\pi}\cdot 2\pi \cdot \frac{1}{(N+1)\operatorname{sen}^2(\delta/2)} \to 0$.
\end{proof}

\begin{observacion}{Norma $L_1$ de $F_N$ y acotación de $\sigma_N(f)$}{obs_norma_fejer}
Como $\frac{1}{2\pi}\int_{-\pi}^{\pi}F_N(t)\,dt = 1$ y $F_N(t) \ge 0$ para todo $t$, se tiene
\[
\frac{1}{2\pi}\int_{-\pi}^{\pi}|F_N(t)|\,dt = 1 \implies \|F_N\|_1 = 1.
\]
Además, por la desigualdad de la norma de la convolución,
\[
\|\sigma_N(f)\|_1 = \|f * F_N\|_1 \le \|f\|_1\,\|F_N\|_1 = \|f\|_1.
\]
\end{observacion}

\begin{nota}{Resumen: propiedades fundamentales de $(F_N)_N$}{resumen_prop_fejer}
Las propiedades que se usarán (y que definen el siguiente concepto) son:
\begin{enumerate}
\item[(1)] $\|F_N\|_1$ acotada (Observación anterior).
\item[(2)] Integra $1$ (Teorema 1.19 (1)).
\item[(3)] Se acumula la masa en el centro (Teorema 1.19 (4)).
\end{enumerate}
\end{nota}

\subsection{Núcleos de sumabilidad}

\begin{definicion}{Núcleo de sumabilidad}{nucleo_sumabilidad}
Una familia de funciones $(K_\varepsilon)_{\varepsilon>0}$ (numerable o no numerable) es un \textbf{núcleo de sumabilidad} sobre $\mathbb{T}$ si:
\begin{enumerate}
\item[(1)] $\displaystyle\sup_{\varepsilon>0}\left( \frac{1}{2\pi}\int_{-\pi}^{\pi}|K_\varepsilon(t)|\,dt \right) = \sup_{\varepsilon>0}\|K_\varepsilon\|_1 < \infty$.
\item[(2)] $\displaystyle\frac{1}{2\pi}\int_{-\pi}^{\pi}K_\varepsilon(t)\,dt = 1$ para todo $\varepsilon > 0$.
\item[(3)] Para cada $0 < \delta < \pi$ fijo se tiene $\displaystyle\lim_{\varepsilon\to 0^+}\left( \frac{1}{2\pi}\int_{\delta<|t|<\pi}|K_\varepsilon(t)|\,dt \right) = 0$.
\end{enumerate}
\end{definicion}

\begin{teorema}{Teorema 1.20}{thm_1_20}
Si $(K_\varepsilon)_{\varepsilon>0}$ es un núcleo de sumabilidad y $f \in C(\mathbb{T})$, entonces
\[
f * K_\varepsilon \xrightarrow{\text{unif}} f.
\]
\end{teorema}

\begin{nota}{Sucesiones como núcleos de sumabilidad}{nota_sucesion_nucleo}
Además, tomando $\varepsilon_N = 1/N$ y $K_N = K_{\varepsilon_N}$ se cubre el caso de una sucesión $(K_N)_N$ (como $(F_N)_N$, donde $N$ juega el papel de $1/\varepsilon$). Por el Teorema 1.19 y la Observación anterior, $(F_N)_N$ cumple (1), (2) y (3), luego es un núcleo de sumabilidad y, por el Teorema 1.20, $\sigma_N(f) \to f$ uniformemente para $f \in C(\mathbb{T})$. El enunciado del teorema ya está en la clase del 15/10; su demostración queda para la siguiente clase (no está en los apuntes).
\end{nota}
